\begin{frame}{One new choice changes the pricing problem}
\begin{block}{The question}
An American option lets its holder choose \emph{when} to exercise. How do we
put a no-arbitrage value on a payoff whose exercise date is itself a decision?
\end{block}
\begin{enumerate}
  \item Separate the right to exercise from the payoff received on exercise.
  \item Turn the exercise date into a stopping-time problem.
  \item Solve the discrete problem by backward induction.
  \item Read the continuous-time version as a free-boundary problem.
  \item Specialize to puts and calls: exercise boundary, and why the call is European.
  \item Compute prices numerically: binomial trees, simulation with regression
        (Longstaff--Schwartz), and finite differences for the PDE.
\end{enumerate}
\end{frame}

\section{The exercise right}

\begin{frame}{European and American options have the same payoff function}
Let \(g(S)=(S-K)^+\) for a call or \(g(S)=(K-S)^+\) for a put.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{European option}
Exercise is fixed at maturity:
\[
  \text{payoff}=g(S_T).
\]
There is no exercise decision before \(T\).
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{American option}
The holder chooses an admissible time
\[
  t\leq\tau\leq T,\qquad \text{payoff}=g(S_\tau).
\]
The payoff map is unchanged; the timing right is valuable.
\end{exampleblock}
\end{columns}
\begin{marketbox}
Maturity is always available. Therefore an American option cannot be worth
less than its European counterpart.
\end{marketbox}
\end{frame}

\begin{frame}{The first inequality is immediate}
For the same strike, underlying, and maturity,
\[
\boxed{V_t^{\mathrm{Am}}\geq V_t^{\mathrm{Eu}}.}
\]
\begin{block}{Why?}
The American holder can choose \(\tau=T\), thereby reproducing the European
payoff. Any better exercise rule can only add value.
\end{block}
\begin{pausebox}
Does \emph{in the money} automatically mean \emph{exercise now} for an
American option?
\end{pausebox}

\begin{alertblock}{No}
An in-the-money option should be exercised only if \(g(S_t)\) is at least
its \emph{continuation value}.
\end{alertblock}
\end{frame}

\begin{frame}{Early exercise has an economic cost and benefit}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Exercise a put now}
Receive \(K-S_t\) and invest the cash. This can be attractive when the put
is deep in the money and interest rates are positive.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Wait instead}
Keep downside insurance and optionality. A further fall in the stock can
increase the eventual payoff.
\end{exampleblock}
\end{columns}
\begin{simplificationbox}
An American call on a non-dividend-paying stock is a special case: under the
usual Black--Scholes assumptions, early exercise is never optimal. A call on
a dividend-paying stock may be different.
\end{simplificationbox}
\end{frame}

\begin{frame}{A stopping time prevents peeking into the future}
At date \(t\), an exercise rule may use only information known at \(t\).
Formally, \(\tau\) is a stopping time if
\[
 \{\tau=t\}\in\cF_t \qquad \text{for every }t.
\]
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{exampleblock}{Admissible rule}
Exercise the put the first time the stock is below \(80\).
\end{exampleblock}
\column{0.48\textwidth}
\begin{alertblock}{Not admissible}
Exercise today if tomorrow's stock will be below \(80\).
\end{alertblock}
\end{columns}
\begin{marketbox}
The word \emph{stopping} encodes the information constraint. It makes the
exercise choice economically meaningful.
\end{marketbox}
\end{frame}

\section{Discrete time: value by backward induction}

\begin{frame}{The American price is an optimal-stopping value}
Work on dates \(0,1,\ldots,T\), with one-period gross rate \(R=1+r\) and
discount factor \(\beta=R^{-1}\). Under a risk-neutral measure \(\Q\),
\[
\boxed{V_t=\operatorname*{sup}_{\tau\in\mathcal T_t}
  \E^\Q\!\left[\beta^{\tau-t}g(S_\tau)\mid\cF_t\right],}
\]
where \(\mathcal T_t\) contains stopping times in \(\{t,\ldots,T\}\).
\begin{block}{Interpretation}
At each node, evaluate every feasible future exercise rule under \(\Q\), then
choose the best one. The rule may adapt to information revealed along the path.
\end{block}
\end{frame}

\begin{frame}{At every node there are exactly two actions}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Exercise}
Stop now and receive intrinsic value
\[
  \boxed{g(S_t).}
\]
No future uncertainty remains.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Continue}
Keep the option for one period:
\[
  \boxed{C_t=\beta\E^\Q[V_{t+1}\mid\cF_t].}
\]
Tomorrow's value already reflects later optimal choices.
\end{exampleblock}
\end{columns}
\begin{marketbox}
Backward induction gives
\[
\boxed{V_T=g(S_T),\qquad V_t=\max\{g(S_t),C_t\}.}
\]
\end{marketbox}
\end{frame}

\begin{frame}{The exercise rule follows the comparison}
\[
\text{exercise at }t
\quad\Longleftrightarrow\quad
g(S_t)\geq \beta\E^\Q[V_{t+1}\mid\cF_t].
\]
\begin{center}
\begin{tabular}{c|c|c}
\toprule
 & Intrinsic \(g(S_t)\) & Continuation \(C_t\)\\
\midrule
Exercise region & \(g(S_t)\geq C_t\) & stop\\
Continuation region & \(g(S_t)<C_t\) & wait\\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{Worked example: a two-period American put}
\begin{modelbox}
\begin{center}
\begin{tabular}{c|c}
\toprule
Initial stock & \(S_0=100\)\\
Up/down factors & \(u=1.2,\quad d=0.8\)\\
Gross risk-free return & \(R=1.05\)\\
Strike & \(K=105\)\\
\bottomrule
\end{tabular}
\end{center}
The risk-neutral up probability is
\[
p=\frac{R-d}{u-d}=\frac{1.05-0.8}{1.2-0.8}=0.625,
\qquad 1-p=0.375.
\]
\end{modelbox}
At date \(1\), stock values are \(120,80\). At maturity, the recombining
tree has values \(144,96,64\).
\end{frame}

\begin{frame}{Step 1: write the terminal payoffs}
For the put \(g(S)=(105-S)^+\):
\begin{center}
\begin{tabular}{c|ccc}
\toprule
\(S_2\) & \(144\) & \(96\) & \(64\)\\
\midrule
\(g(S_2)\) & \(0\) & \(9\) & \(41\)\\
\bottomrule
\end{tabular}
\end{center}
\begin{block}{The node with \(S_1=120\)}
Immediate exercise gives \(0\). Continuing gives
\[
C_1^{(120)}=\frac{0.625(0)+0.375(9)}{1.05}=3.21.
\]
So we continue, and \(V_1^{(120)}=3.21\).
\end{block}
\end{frame}

\begin{frame}{Step 2: is early exercise optimal at \(S_1=80\)?}
\begin{pausebox}
At \(S_1=80\), the immediate put payoff is \(105-80=25\). The two terminal
payoffs after waiting are \(9\) and \(41\). Should the holder exercise or
continue?
\end{pausebox}

\begin{exampleblock}{Compare the two values}
\[
g(80)=25,\qquad
C_1^{(80)}=\frac{0.625(9)+0.375(41)}{1.05}=20.
\]
Since \(25>20\), exercise at this node and set \(V_1^{(80)}=25\).
\end{exampleblock}
\begin{marketbox}
The optimal action depends on the node. The American put has an exercise
region, not a pre-announced exercise date.
\end{marketbox}
\end{frame}

\begin{frame}{Step 3: return to time zero}
At \(t=0\), immediate exercise is worth \(g(100)=5\). Continuing is worth
\[
C_0=\frac{0.625(3.21)+0.375(25)}{1.05}=10.84.
\]
Therefore
\[
\boxed{V_0^{\mathrm{Am}}=\max\{5,10.84\}=10.84.}
\]
\begin{block}{The policy}
Wait at time \(0\); at time \(1\), exercise only in the down node \(S_1=80\).
The price and the policy are computed together.
\end{block}
\end{frame}

\begin{frame}{The tree records both the exercise policy and the hedge}
\begin{center}
\begin{tabular}{c|c|c|c|c}
\toprule
Date & Stock & Exercise value & Continuation value & Decision\\
\midrule
\(0\) & \(100\) & \(5\) & \(10.84\) & continue\\
\(1\), up & \(120\) & \(0\) & \(3.21\) & continue\\
\(1\), down & \(80\) & \(25\) & \(20\) & exercise\\
\bottomrule
\end{tabular}
\end{center}
At time \(0\), the one-period replicating hedge for the two future option
values \(3.21\) and \(25\) uses
\[
\Delta_0=\frac{3.214-25}{120-80}=-0.545,\qquad
B_0\approx65.31.
\]
\begin{marketbox}
\(\Delta_0S_0+B_0=10.84\). The hedge finances the option while it continues;
at the down node it instead finances the exercise payment \(25\).
\end{marketbox}
\end{frame}

\begin{frame}{The early-exercise premium is visible in the numbers}
If exercise were allowed only at maturity, the European put price would be
\[
V_0^{\mathrm{Eu}}
=\frac{2(0.625)(0.375)(9)+(0.375)^2(41)}{1.05^2}
=9.06.
\]
Hence
\[
\boxed{V_0^{\mathrm{Am}}-V_0^{\mathrm{Eu}}=10.84-9.06=1.78.}
\]
\begin{alertblock}{What creates the difference?}
Not a different terminal payoff. It is the right to take \(25\) at the down
node rather than be forced to carry the position to maturity.
\end{alertblock}
\end{frame}

\begin{frame}{A fast decision check}
\begin{pausebox}
At a node of an American put, intrinsic value is \(18\) and discounted
risk-neutral continuation value is \(21\). What is the action and what is the
option value at the node?
\end{pausebox}

\begin{exampleblock}{Answer}
Continue. The option is worth \(\max\{18,21\}=21\). Being in the money is
not enough: exercising would throw away \(3\) of continuation value.
\end{exampleblock}
\end{frame}

\begin{frame}{The Snell envelope is the discounted value process}
\small
Define discounted exercise payoff and value by
\[
Y_t=\beta^t g(S_t),\qquad \widetilde V_t=\beta^tV_t.
\]
The recursion becomes
\[
\widetilde V_T=Y_T,\qquad
\widetilde V_t=\max\bigl\{Y_t,\E^\Q[\widetilde V_{t+1}\mid\cF_t]\bigr\}.
\]
\begin{block}{Snell envelope}
\(\widetilde V\) is the smallest \(\Q\)-supermartingale that dominates the
discounted payoff process \(Y\). Supermartingale:
\(\E^\Q[\widetilde V_{t+1}\mid\cF_t]\leq\widetilde V_t\), i.e.\ it decreases
on average; here this follows directly from the \(\max\).
\end{block}
\begin{marketbox}
It is the American analogue of risk-neutral valuation: the minimal
discounted process that can always cover an exercise request. The next
slide shows this is \emph{both} the buyer's optimal-stopping value and the
seller's minimal hedging cost --- the same number, seen from two sides.
\end{marketbox}
\end{frame}

\begin{frame}{The seller must superhedge every admissible exercise date}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Buyer's value}
The recursion \emph{is} the buyer's best achievable outcome:
\[
V_t=\max\{g(S_t),\beta\E^\Q[V_{t+1}\mid\cF_t]\}.
\]
No stopping rule can do better for the holder.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Seller's price}
The seller does not know which \(\tau\) the buyer will pick. A
self-financing \(X^\pi\) must satisfy
\[
  X^\pi_u\geq g(S_u)
  \qquad\text{for every admissible }u\geq t.
\]
\end{exampleblock}
\end{columns}
The no-arbitrage price is the cheapest such protection:
\[
\boxed{V_t^{\mathrm{Am}}
=\inf\left\{x:\exists\,\pi\text{ self-financing with }
X_u^\pi\geq g(S_u)\text{ for every }u\geq t\right\}.}
\]
\begin{marketbox}
Tracking \(V_t\) with a replicating trade already superhedges every node, so
it costs no more than \(V_t\); no cheaper strategy survives every stopping
choice the buyer might make. Seller's minimal capital = buyer's optimal
value.
\end{marketbox}
\end{frame}

\section{Continuous time: an obstacle and a boundary}

\begin{frame}{In Black--Scholes, the choice is over all future times}
\small
Under \(\Q\), suppose
\[
\dd S_t=rS_t\dd t+\sigma S_t\dd W_t^{\Q}.
\]
\begin{block}{Why work directly under \(\Q\)?}
Fix a stopping time \(\tau\). Receiving \(g(S_\tau)\) at \(\tau\) is a claim
whose no-arbitrage price is \(\E^\Q[e^{-r(\tau-t)}g(S_\tau)\mid\cF_t]\)
(Lectures~6--8). The holder picks the best \(\tau\), so the American price is
the supremum of these prices over \(\tau\), as in the discrete recursion.
Only the \(\Q\)-dynamics of \(S\) matter: drift \(r\), not \(\mu\).
\end{block}
For an American payoff \(g\), its value function is
\[
\boxed{v(t,s)=\sup_{\tau\in\mathcal T_{t,T}}
\E^\Q\!\left[e^{-r(\tau-t)}g(S_\tau)\mid S_t=s\right].}
\]
\textbf{Exercise region} \(\{v=g\}\): take the payoff now.\quad
\textbf{Continuation region} \(\{v>g\}\): keep the option alive.
\end{frame}

\begin{frame}{What does \(\E^\Q_{t,s}\) mean?}
\small
Start the stock at time \(t\) with \(S_t=s\), and use its \(\Q\)-dynamics.
For a future payoff \(h(S_T)\),
\[
\boxed{\E^\Q_{t,s}[h(S_T)]
=\E^\Q[h(S_T)\mid S_t=s].}
\]
\textbf{Markov property:} the current stock price summarizes the relevant past.
\[
\E^\Q[h(S_T)\mid\cF_t]=F(t,S_t),\qquad
F(t,s)=\E^\Q_{t,s}[h(S_T)].
\]
\begin{exampleblock}{Example: the Black--Scholes stock, without dividends}
\[
\E^\Q_{t,s}[S_T]=s\,e^{r(T-t)},\qquad
\E^\Q_{t,s}[e^{-r(T-t)}S_T]=s.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{At \((t,s)\), the holder has two options}
\small
Write \(\mathcal Av=rs\,v_s+\tfrac12\sigma^2s^2v_{ss}-rv\), and assume
(formally) that \(v\) is smooth enough for It\^o. Then
\[
\dd\bigl(e^{-ru}v(u,S_u)\bigr)
=e^{-ru}(v_t+\mathcal Av)(u,S_u)\dd u+e^{-ru}\sigma S_uv_s(u,S_u)\dd W_u^\Q .
\]
Integrate over \([t,t+\dd t]\) and take \(\E^\Q_{t,s}\); the \(\dd W\) term has
zero mean:
\[
\E^\Q_{t,s}\!\left[e^{-r\dd t}v(t+\dd t,S_{t+\dd t})\right]
=v(t,s)+(v_t+\mathcal Av)(t,s)\dd t+o(\dd t).
\]
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Option 1: exercise now}
Stop at \(\tau=t\) and receive \(g(s)\).
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Option 2: wait \(\dd t\), then act optimally}
Worth \(\E^\Q_{t,s}\!\left[e^{-r\dd t}v(t+\dd t,S_{t+\dd t})\right]\).
\end{exampleblock}
\end{columns}
\medskip
The holder takes the better option:
\(v(t,s)=\max\{\text{Option 1},\text{Option 2}\}\), formally.
Each option gives an \textbf{inequality}; the optimal one gives an
\textbf{equality}.
\end{frame}

\begin{frame}{Option 1, exercise now: the value lies above the payoff}
Exercising at \(\tau=t\) uses only the current state, so it is an admissible
stopping time. Its value is \(g(s)\), hence
\[
\boxed{v(t,s)\geq g(s)\qquad\text{for every }(t,s).}
\]
\begin{block}{Why this must hold in a market}
If an American put traded below \(K-s\), buy it and exercise immediately:
the cash received exceeds the price paid. This is an arbitrage.
\end{block}
\begin{marketbox}
Equality \(v=g\) holds exactly where exercising now is optimal. This splits
the \((t,s)\) plane into the \textbf{exercise region} \(\{v=g\}\) and the
\textbf{continuation region} \(\{v>g\}\).
\end{marketbox}
\end{frame}

\begin{frame}{Option 2, wait \(\dd t\): the value cannot grow faster than \(r\)}
\small
Waiting for \(\dd t\), then acting optimally, is always admissible:
\[
v(t,s)\geq\E^\Q_{t,s}\!\left[e^{-r\dd t}v(t+\dd t,S_{t+\dd t})\right]
=v(t,s)+(v_t+\mathcal Av)(t,s)\dd t+o(\dd t).
\]
Subtract \(v(t,s)\), divide by \(\dd t\), and let \(\dd t\downarrow0\):
\[
\boxed{v_t+\mathcal Av\leq0\qquad\text{everywhere, including the exercise region.}}
\]
\begin{block}{Interpretation: a supermartingale}
An adapted, integrable process \(X\) is a \(\Q\)-\textbf{supermartingale} if
\(\E^\Q[X_u\mid\cF_t]\leq X_t\) for all \(t\leq u\): it decreases on average
(a martingale is the case of equality). By It\^o, \(e^{-rt}v(t,S_t)\) has
drift \(e^{-rt}(v_t+\mathcal Av)\leq0\), so it is a \(\Q\)-supermartingale:
in discounted terms, holding the option loses value on average.
\end{block}
\textbf{Check on the put}, where \(v=g=K-s\):
\(g_t+\mathcal Ag=rs(-1)-r(K-s)=-rK<0\). Waiting in the exercise region
loses the interest \(rK\dd t\) on the strike.
\end{frame}

\begin{frame}{In the continuation region, waiting is optimal: the BS PDE}
Suppose \(v(t,s)>g(s)\). Exercising now is \emph{strictly} worse than holding.
\begin{block}{The holder waits over \([t,t+\dd t]\)}
\(v-g\) is continuous and the path is continuous. For small \(\dd t\), the
path stays where \(v>g\), so stopping in \([t,t+\dd t]\) is never optimal.
Option 2 attains the maximum:
\[
v(t,s)=\E^\Q_{t,s}\!\left[e^{-r\dd t}v(t+\dd t,S_{t+\dd t})\right]+o(\dd t)
\quad\Longrightarrow\quad
\boxed{v_t+\mathcal Av=0\quad\text{where }v>g.}
\]
\end{block}
\begin{marketbox}
This is the Black--Scholes PDE. Until exercise, \(e^{-rt}v(t,S_t)\) is a
martingale, and the American option is hedged like a European one.
\end{marketbox}
\begin{simplificationbox}
This is a formal argument: at the exercise boundary, \(v\) is only \(C^1\) in
\(s\). A rigorous proof uses the Snell envelope or viscosity solutions.
\end{simplificationbox}
\end{frame}

\begin{frame}{The PDE has an obstacle}
At each \((t,s)\), Options 1 and 2 give two inequalities, and the optimal
option turns one of them into an equality:
\[
\boxed{\max\left\{g(s)-v,\;
v_t+rs\,v_s+\tfrac12\sigma^2s^2v_{ss}-rv\right\}=0,
\qquad v(T,s)=g(s).}
\]
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Continuation region \(\{v>g\}\)}
\(v_t+\mathcal Av=0\): the option behaves like a European claim.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Exercise region \(\{v=g\}\)}
\(v=g\), and Option 2 still requires \(g_t+\mathcal Ag\leq0\).
For a put with \(s<K\): \(\mathcal Ag=-rK<0\). Waiting costs the
interest on \(K\).
\end{exampleblock}
\end{columns}
\begin{simplificationbox}
Unlike a European option, the region where the PDE holds is not known in
advance. Finding it is part of the pricing problem.
\end{simplificationbox}
\end{frame}

\section{American puts and calls}

\begin{frame}{From now on: vanilla puts and calls}
So far \(g\) was any payoff. We now specialize to
\(g(s)=(K-s)^+\) and \(g(s)=(s-K)^+\), with no dividends unless stated.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{American put}
Early exercise can be optimal. There is no closed-form price: the value
\(v\) and the exercise boundary \(s^*(t)\) must be computed numerically.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{American call, no dividends}
Early exercise is never optimal, so the American call equals the European
call: the Black--Scholes formula applies.
\end{exampleblock}
\end{columns}
\begin{block}{What ``computing'' means here}
Pricing the put means solving its obstacle problem: by backward induction on
a binomial tree, by Monte Carlo simulation with regression
(Longstaff--Schwartz), or by discretizing the PDE (finite differences). This
is the next section.
\end{block}
\end{frame}

\begin{frame}{The put PDE needs terminal and spatial conditions}
Assume no dividends, \(r\geq0\), and \(g(s)=(K-s)^+\). Recall
\[
\mathcal Av=rs\,v_s+\tfrac12\sigma^2s^2v_{ss}-rv.
\]
The obstacle problem can be written as three conditions:
\[
\boxed{v-g\geq0,\qquad -(v_t+\mathcal Av)\geq0,\qquad
       (v-g)(v_t+\mathcal Av)=0.}
\]
\begin{block}{Data for the backward calculation}
\[
v(T,s)=(K-s)^+,\qquad v(t,0)=K,\qquad
\lim_{s\to\infty}v(t,s)=0.
\]
On a finite grid, choose \(S_{\max}>K\) and approximate
\(v(t,S_{\max})=0\). Check the effect of increasing \(S_{\max}\).
\end{block}
\textbf{Continue:} \(v>g\) forces the PDE equality.
\quad\textbf{Exercise:} \(v=g\).
\end{frame}

\begin{frame}{For an American put, the boundary is the decision rule}
For a put, there is typically a boundary \(s^*(t)\) such that
\[
\begin{cases}
\text{exercise}, & s\leq s^*(t),\\
\text{continue}, & s>s^*(t).
\end{cases}
\]
At the boundary, the value joins the payoff smoothly:
\[
\boxed{v(t,s^*(t))=K-s^*(t),\qquad v_s(t,s^*(t))=-1.}
\]
\begin{block}{Boundary conditions}
At maturity, \(v(T,s)=(K-s)^+\). For positive rates, the put boundary
typically approaches \(K\) from below as maturity approaches.
\end{block}
\end{frame}

\begin{frame}{Why a non-dividend-paying American call is European}
Let \(C^{\mathrm{Eu}}(t,s)\) be a European call and suppose \(r\geq0\), with
no dividends. For every \(t<T\),
\[
C^{\mathrm{Eu}}(t,s)\geq s-Ke^{-r(T-t)}\geq s-K.
\]
Thus the European call already dominates the value from immediate exercise,
\((s-K)^+\).
\begin{marketbox}
Waiting until maturity is feasible for the American holder, and immediate
exercise never improves on the European value. Therefore
\[
\boxed{C^{\mathrm{Am}}=C^{\mathrm{Eu}}\quad\text{(no dividends).}}
\]
\end{marketbox}
\end{frame}

\begin{frame}{Dividends change the call trade-off}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{No dividend}
The option holder does not receive a stock dividend. Giving up the option to
buy the stock early has no compensating cash-flow benefit.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Known dividend}
Exercising before an ex-dividend date can give ownership of the stock and
therefore the dividend. Early exercise of a call can become optimal.
\end{exampleblock}
\end{columns}
\begin{pausebox}
Which object changes with dividends: the payoff, stopping-time logic, or
the continuation value?
\end{pausebox}

\begin{exampleblock}{Answer}
The payoff and stopping-time logic remain. Dividends alter the economics of
waiting, hence the continuation value and the exercise boundary.
\end{exampleblock}
\end{frame}

\section{Computing American prices}

\begin{frame}{Three numerical routes, three different approximations}
\small
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}p{0.17\textwidth}
                >{\raggedright\arraybackslash}p{0.30\textwidth}
                >{\raggedright\arraybackslash}p{0.40\textwidth}}
\toprule
Method & What is approximated & How the continuation value is obtained\\
\midrule
\textbf{Binomial tree}
  & The \emph{stock model}: \(S\) moves to \(uS\) or \(dS\) at dates
    \(0,\Delta t,\ldots,T\)
  & Exactly, from the two children:
    \(C_n=e^{-r\Delta t}[pV_{n+1}^{\mathrm{up}}+(1-p)V_{n+1}^{\mathrm{down}}]\)\\
\midrule
\textbf{Longstaff--Schwartz}
  & Only the \emph{exercise dates}: paths of the Black--Scholes model are
    simulated; exercise is allowed at \(t_0,\ldots,t_N\)
  & Estimated, by regressing realized cash flows on functions of \(S_{t_k}\)
    across simulated paths\\
\midrule
\textbf{Finite differences}
  & The \emph{PDE}: \(\max\{g-v,\;v_t+\mathcal Av\}=0\) on an \((s,t)\) grid
  & Through the discretized PDE operator, then the obstacle \(V\geq g\)\\
\bottomrule
\end{tabular}
\end{center}
\textbf{Common thread:} compare exercise with continuation, then refine
the exercise grid and control each method's space, sampling, or regression errors.
\end{frame}

\subsection{The binomial model: backward induction on a tree}

\begin{frame}{A binomial tree is backward induction made numerical}
With \(N\) steps of length \(\Delta t=T/N\), the CRR parameters are
\[
u=e^{\sigma\sqrt{\Delta t}},\quad d=u^{-1},\quad
p=\frac{e^{r\Delta t}-d}{u-d}.
\]
At every node \((n,j)\), compute
\[
\boxed{V_n^j=\max\left\{g(S_n^j),e^{-r\Delta t}
\left[pV_{n+1}^{j+1}+(1-p)V_{n+1}^j\right]\right\}.}
\]
\begin{block}{Output}
The tree supplies both a price and a discrete approximation to the exercise
boundary: mark every node at which the maximum selects \(g(S_n^j)\).
\end{block}
\end{frame}

\subsection{Simulation: the Longstaff--Schwartz method}

\begin{frame}{Simulation uses the Black--Scholes model, not a tree}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Binomial tree}
\(S\) can only move to \(uS\) or \(dS\). Paths recombine, so a node is
reached by many paths, and its continuation value is an exact average over
its two children.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Monte Carlo simulation}
Draw \(M\) paths from the Black--Scholes \(\Q\)-dynamics, exactly:
\[
S_{t_{k+1}}=S_{t_k}\exp\Bigl(\bigl(r-\tfrac12\sigma^2\bigr)\Delta t
+\sigma\sqrt{\Delta t}\,Z_{k+1}\Bigr),
\]
with \(Z_{k+1}\sim\mathcal N(0,1)\) i.i.d. Paths do not recombine: each state
is visited once.
\end{exampleblock}
\end{columns}
\begin{marketbox}
Nothing is binomial here. The only discrete feature is the set of exercise
dates \(0=t_0<\cdots<t_N=T\): Longstaff--Schwartz prices a \emph{Bermudan}
option, which approaches the American price as the exercise grid is refined.
\end{marketbox}
\begin{simplificationbox}
No siblings to average over: \(C_k(s)\) must be \emph{estimated}.
\end{simplificationbox}
\end{frame}

\begin{frame}{Why plain Monte Carlo is not enough}
For a European option, simulation estimates
\[
e^{-rT}\E^\Q[g(S_T)]
\]
by averaging terminal discounted payoffs.
\begin{alertblock}{The American difficulty}
At an intermediate state, the relevant object is
\[
\E^\Q[\text{future cash flow}\mid S_t=s],
\]
the conditional continuation value. Simulated paths do not reveal this
function directly at a single node.
\end{alertblock}
\begin{marketbox}
American pricing is not just simulation; it is simulation plus a way to
estimate the decision rule.
\end{marketbox}
\end{frame}

\begin{frame}{Longstaff--Schwartz: the setting and the key idea}
\small
\begin{block}{Ingredients}
\begin{itemize}
\item Exercise dates \(0=t_0<t_1<\cdots<t_N=T\): exercise only at these dates (Bermudan).
\item \(M\) independent paths \(S^{(m)}_{t_0},\ldots,S^{(m)}_{t_N}\) simulated
      from the Black--Scholes \(\Q\)-dynamics (no tree), \(m=1,\ldots,M\).
\item Basis functions \(\psi_0,\ldots,\psi_L\), e.g.\ \(1,s,s^2\) or Laguerre polynomials.
\item Per path, a running \textbf{cash flow} \(CF^{(m)}\) and its
      \textbf{date} \(\tau^{(m)}\) under the current rule.
\end{itemize}
\end{block}
\begin{exampleblock}{Key idea: least-squares Monte Carlo (LSMC)}
With \(\tau\) the optimal exercise date after \(t_k\), the continuation value
is a function of the state:
\[
C_k(s)=\E^\Q\!\left[e^{-r(\tau-t_k)}g(S_\tau)\mid S_{t_k}=s\right]
\approx\sum_{\ell=0}^L\beta_{k,\ell}\,\psi_\ell(s).
\]
The \(\beta_{k,\ell}\) are fitted \emph{across paths}: each path gives one point
\(\bigl(S^{(m)}_{t_k},\ \text{its realized discounted cash flow}\bigr)\).
\end{exampleblock}
\end{frame}

\begin{frame}{LSMC (1/3): a conditional mean is a function of the state}
\small
At one fixed date \(t_k\), keep the later exercise rule fixed and write
\[
X=S_{t_k},\qquad Y=e^{-r(\tau-t_k)}g(S_\tau),\qquad \tau>t_k.
\]
Each simulated path supplies one pair \((X^{(m)},Y^{(m)})\).
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{An unconditional average}
\[
\frac1M\sum_{m=1}^M Y^{(m)}\approx\E^\Q[Y].
\]
One number: it pools all current stock prices.
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{The conditional mean we need}
\[
c(s)=\E^\Q[Y\mid X=s].
\]
A function: what is the mean future cash flow \emph{at this current state}?
\end{exampleblock}
\end{columns}
\begin{exampleblock}{Pool information across paths}
For continuous \(X\), paths almost surely have different current prices.
Regression fits a common function across the cloud of pairs \((X,Y)\).
\end{exampleblock}
\end{frame}

\begin{frame}{LSMC (2/3): from conditional expectation to regression}
\small
For \(Y\in L^2\), the conditional mean solves
\[
\boxed{c=\operatorname*{arg\,min}_{f:\,f(X)\in L^2}
\E^\Q[(Y-f(X))^2],}
\qquad c(X)=\E^\Q[Y\mid X].
\]
\textbf{Restrict \(f\) to a finite basis}, for example \(1,s,s^2\):
\[
f_\beta(s)=\sum_{\ell=0}^L\beta_\ell\psi_\ell(s),\qquad
\beta\in\mathbb R^{L+1}.
\]
\textbf{Fit the coefficients from simulated pairs:}
\[
\widehat\beta=\operatorname*{arg\,min}_{\beta}\frac1M\sum_{m=1}^M
\left(Y^{(m)}-f_\beta(X^{(m)})\right)^2.
\]
\[
\boxed{\widehat C_k(s)=f_{\widehat\beta}(s).}
\]
\end{frame}

\begin{frame}{LSMC (3/3): the actual matrix in the eight-path example}
\small
Put with \(K=1.10\), \(r=0.06\), dates \(0,1,2,3\); work at \(t=2\).
The in-the-money paths are \(1,3,4,6,7\), and the later rule is exercise at \(3\).
\[
\Psi=\begin{pmatrix}

1&1.07&1.1449\\
1&0.97&0.9409\\
1&0.77&0.5929\\
1&0.84&0.7056
\end{pmatrix},\quad
y=e^{-0.06}\begin{pmatrix}0\\0.07\\0.18\\0.20\\0.09\end{pmatrix},\quad
\widehat\beta\approx\begin{pmatrix}-1.069988\\2.983411\\-1.813576\end{pmatrix}.
\]
\begin{exampleblock}{From coefficients to a decision}
At \(s=0.97\), evaluate the fitted function:
\[
\widehat C_2(0.97)\approx0.1175
\quad<\quad g(0.97)=0.13\quad\Longrightarrow\quad\text{exercise}.
\]
\end{exampleblock}
The response \(y\) is already discounted to \(t=2\); do not discount the fit again.
\end{frame}

\begin{frame}{Longstaff--Schwartz: the algorithm}
\small
\begin{enumerate}
\item \textbf{Initialize at maturity.} For every path,
      \(CF^{(m)}=g(S^{(m)}_T)\) and \(\tau^{(m)}=T\).
\item \textbf{Backward loop.} For \(k=N-1,N-2,\ldots,1\):
  \begin{enumerate}
  \item[(a)] \emph{Select} the in-the-money paths
        \(I_k=\{m:g(S^{(m)}_{t_k})>0\}\).
  \item[(b)] For \(m\in I_k\), the realized discounted cash flow
        \(Y^{(m)}=e^{-r(\tau^{(m)}-t_k)}CF^{(m)}\).
  \item[(c)] \emph{Regress} by least squares:
        \[
        \widehat\beta_k=\arg\min_\beta\sum_{m\in I_k}
        \Bigl(Y^{(m)}-\sum_{\ell=0}^L\beta_\ell\,\psi_\ell(S^{(m)}_{t_k})\Bigr)^2,
        \qquad
        \widehat C_k(s)=\sum_{\ell}\widehat\beta_{k,\ell}\,\psi_\ell(s).
        \]
  \item[(d)] \emph{Decide and update:} for \(m\in I_k\), if
        \(g(S^{(m)}_{t_k})>\widehat C_k(S^{(m)}_{t_k})\), set
        \(CF^{(m)}\leftarrow g(S^{(m)}_{t_k})\) and \(\tau^{(m)}\leftarrow t_k\).
        Otherwise keep the later cash flow.
  \end{enumerate}
\item \textbf{Price at \(t_0\).} All paths share \(S_0\), so no regression is
      needed:
      \[
      \boxed{\widehat V_0=\max\Bigl\{g(S_0),\ \frac1M\sum_{m=1}^M
      e^{-r\tau^{(m)}}CF^{(m)}\Bigr\}.}
      \]
\end{enumerate}
\end{frame}

\begin{frame}{Longstaff--Schwartz: why each design choice}
\small
\begin{block}{Regress realized cash flows, not estimated values}
\(Y^{(m)}\) is what the path actually pays under the current rule. The
regression only \emph{selects} the rule; the price averages real cash flows,
so errors affect the exercise policy, including earlier regression targets.
\end{block}
\begin{block}{Use only in-the-money paths}
Out of the money, \(g=0\) and exercise is never chosen. Fitting only where
the decision matters improves the fit with few basis functions.
\end{block}
\begin{alertblock}{Evaluate a fixed rule on fresh paths}
Its expected discounted payoff is at most the Bermudan value; an individual
sample estimate can exceed it. Reusing training paths can create upward
foresight bias. Fit the rule on one set, then price on independent paths.
\end{alertblock}
Accuracy depends on \(M\), the basis, and the number of exercise dates.
Duality methods (Andersen--Broadie) give an upper bound.
\end{frame}

\subsection{The continuous model: solving the obstacle PDE}

\begin{frame}{Finite differences: build the grid and start at maturity}
For the no-dividend put, with \(r\geq0\), set
\[
s_i=i\Delta s,\quad \Delta s=S_{\max}/M,\qquad
t_n=n\Delta t,\quad \Delta t=T/N,\qquad V_i^n\approx v(t_n,s_i).
\]
Start with \(V_i^N=g_i=(K-s_i)^+\) and work
\textbf{backward}: \(n=N-1,\ldots,0\).
\begin{block}{Centered differences at an interior stock node}
\[
v_s(t_n,s_i)\approx\frac{V_{i+1}^n-V_{i-1}^n}{2\Delta s},\qquad
v_{ss}(t_n,s_i)\approx
\frac{V_{i+1}^n-2V_i^n+V_{i-1}^n}{(\Delta s)^2}.
\]
\end{block}
\begin{block}{Boundary values and the obstacle}
At every time level impose \(V_0^n=K\), \(V_M^n=0\).
The obstacle is \(V_i^n\geq g_i\); the price is read at \(t_0=0\).
\end{block}
\end{frame}

\begin{frame}{Explicit scheme: a continuation step, then the maximum}
\small
Discretizing \(v_t+\mathcal Av=0\) with spatial terms at the known
level \(n+1\) gives, for \(i=1,\ldots,M-1\),
\[
\boxed{C_i^n=a_iV_{i-1}^{n+1}+b_iV_i^{n+1}+c_iV_{i+1}^{n+1},
\qquad V_i^n=\max\{g_i,C_i^n\}.}
\]
Here \(s_i=i\Delta s\), so
\[
\begin{aligned}
a_i&=\tfrac{\Delta t}{2}(\sigma^2i^2-ri),&
c_i&=\tfrac{\Delta t}{2}(\sigma^2i^2+ri),\\
b_i&=1-\Delta t(\sigma^2i^2+r),&
a_i+b_i+c_i&=1-r\Delta t.
\end{aligned}
\]
\begin{alertblock}{Check the weights before using this explicit scheme}
Require \(a_i,b_i,c_i\geq0\) at every interior node. This holds if
\[
\sigma^2i^2\geq ri,\qquad
\Delta t\leq\frac{1}{\sigma^2(M-1)^2+r}.
\]
If a spatial weight is negative, change the drift stencil; reducing
\(\Delta t\) alone cannot fix its sign.
\end{alertblock}
\end{frame}

\begin{frame}{Explicit scheme: the algorithm, step by step}
\small
\begin{enumerate}
\item \textbf{Grid and check.} Choose \(S_{\max}\), \(M\), \(N\); set
      \(\Delta s=S_{\max}/M\), \(\Delta t=T/N\). Compute \(a_i,b_i,c_i\) for
      \(i=1,\ldots,M-1\) and check that all are \(\geq0\); otherwise increase \(N\).
\item \textbf{Initialize at maturity.} For \(i=0,\ldots,M\):
      \(g_i=(K-s_i)^+\) and \(V_i^N=g_i\).
\item \textbf{Backward loop.} For \(n=N-1,N-2,\ldots,0\):
  \begin{enumerate}
  \item[(a)] \emph{Boundaries:} \(V_0^n=K\), \(V_M^n=0\).
  \item[(b)] \emph{Continuation:} for \(i=1,\ldots,M-1\), using only the known level \(n+1\),
        \[
        C_i^n=a_iV_{i-1}^{n+1}+b_iV_i^{n+1}+c_iV_{i+1}^{n+1}.
        \]
  \item[(c)] \emph{Decide:} \(V_i^n=\max\{g_i,C_i^n\}\). If \(g_i\geq C_i^n\)
        and \(g_i>0\), flag node \((n,i)\) as \emph{exercise}.
  \end{enumerate}
\item \textbf{Output.} Interpolate \(V^0\) at \(s=S_0\) for the price. At each
      \(t_n\), the largest flagged \(s_i\) approximates \(s^*(t_n)\).
\end{enumerate}
\begin{block}{Practical notes}
\(O(M)\) per step, no linear solve: the \(\max\) is applied node by node, as on
a tree. Cost of stability: \(\Delta t\propto(\Delta s)^2\), so \(N\) grows like \(M^2\).
\end{block}
\end{frame}
