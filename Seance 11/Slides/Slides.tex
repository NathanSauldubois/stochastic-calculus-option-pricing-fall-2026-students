\begin{frame}{A new model must solve an identifiable problem}

    \begin{enumerate}
    \item Recover what vanilla prices tell us about the terminal distribution.
    \item Construct a one-factor diffusion consistent with a call surface.
    \item Choose pricing dynamics and hedge the variance risk with an option.
    \item Connect Heston parameters, smile shapes, calibration, and simulation.
    \end{enumerate}
    \begin{keybox}One common framework: no dividends, a constant rate \(r\),
    European payoffs, and dynamics specified under a pricing measure \(\Q\).
    \end{keybox}
\end{frame}

\section{1. What the smile asks us to explain}

\begin{frame}{A price quote becomes an implied volatility}

    \[C^{\rm mkt}(K,T)=C^{\rm BS}(S_0,K,T,r,\sigma_{\rm imp}(K,T)).\]
    \begin{block}{Under a constant-volatility Black--Scholes model}
    All strikes and maturities would return the same implied volatility.
    \end{block}
    \begin{pausebox}A one-year put at strike 80 has 28\% implied volatility;
    an ATM call has 20\%. Can one constant \(\sigma\) fit both?\end{pausebox}
    \begin{answerbox}No. Using a different BS input for every option quotes
    prices conveniently; it does not specify one joint model for the stock.\end{answerbox}
\end{frame}

\section{2. Local volatility: from prices to a diffusion}

\begin{frame}{Keep one factor; let its volatility depend on spot}

    \[\boxed{\dd S_t=rS_t\dd t+\sigma_{\rm loc}(t,S_t)S_t\dd W_t^\Q.}\]
    \begin{itemize}
    \item \(\sigma_{\rm loc}(t,s)\) is a deterministic function, evaluated at a random spot.
    \item The current state is still \((t,S_t)\).
    \item Black--Scholes corresponds to \(\sigma_{\rm loc}(t,s)\equiv\sigma\).
    \end{itemize}
    \begin{pausebox}Has the number of independent Brownian risks increased?\end{pausebox}
    \begin{answerbox}No. There is still one stock diffusion shock.\end{answerbox}
\end{frame}

\begin{frame}{Apply the same It\^o and replication argument}

    For a smooth value \(u(t,S_t)\), It\^o gives
    \[\dd u=\left(u_t+rsu_s+\tfrac12\sigma_{\rm loc}^2(t,s)s^2u_{ss}\right)\dd t
      +\sigma_{\rm loc}(t,s)s u_s\dd W_t.\]
    Match the Brownian term with \(\Delta=u_s\), then use self-financing:
    \[\boxed{u_t+rsu_s+\tfrac12\sigma_{\rm loc}^2(t,s)s^2u_{ss}-ru=0,}
       \qquad u(T,s)=g(s).\]
    \begin{keybox}Under the usual regularity and nondegeneracy assumptions,
    the model retains the one-factor replication structure.\end{keybox}
\end{frame}

% BEGIN INLINE: Dupire_Notation
\begin{frame}{Dupire: which variables are we differentiating?}
\small
Fix today's stock \(S_0\), the rate \(r\), and the model coefficient
\(\sigma_{\rm loc}(t,s)\).
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{One option, changing valuation state}
\[
u(t,s;T,K).
\]
Fix the contract \((T,K)\); vary \((t,s)\).
The backward pricing PDE uses \(u_t,u_s,u_{ss}\).
\end{block}
\column{0.48\textwidth}
\begin{exampleblock}{Today's prices of different contracts}
\[
C(T,K)=u(0,S_0;T,K).
\]
Fix \((0,S_0)\); vary \((T,K)\).
The Dupire equation uses \(C_T,C_K,C_{KK}\).
\end{exampleblock}
\end{columns}
\[
p(T,x)\dd x=\Q(S_T\in\dd x\mid S_0),\qquad
C(T,K)=e^{-rT}\int_K^\infty(x-K)p(T,x)\dd x.
\]
The integration variable \(x\) is a future stock price; \(K\) is the strike.
\end{frame}
% END INLINE: Dupire_Notation

\begin{frame}{Call prices encode tail probabilities}

    Keep \(S_0\) fixed and write today's call surface as \(C(T,K)\).
    If \(S_T\) has density \(p(T,s)\),
    \[C(T,K)=e^{-rT}\int_K^\infty(s-K)p(T,s)\dd s.\]
    \begin{pausebox}Differentiate once with respect to strike.\end{pausebox}

    \[\boxed{C_K(T,K)=-e^{-rT}\int_K^\infty p(T,s)\dd s
                  =-e^{-rT}\Q(S_T>K).}\]
    \begin{keybox}A tight call spread approximates a digital payoff.\end{keybox}
\end{frame}

% BEGIN INLINE: Dupire_Strike_Derivatives
\begin{frame}{Strike derivatives: include the moving integration limit}
\small
The density \(p(T,x)\) does not depend on the contract's strike \(K\).
Leibniz's rule gives
\[
\begin{aligned}
C_K(T,K)
&=e^{-rT}\left[-(K-K)p(T,K)
+\int_K^\infty\partial_K(x-K)p(T,x)\dd x\right]\\
&=-e^{-rT}\int_K^\infty p(T,x)\dd x.
\end{aligned}
\]
The lower-limit term is zero because the call payoff vanishes at \(x=K\).
Differentiate the remaining tail integral:
\[
\boxed{C_{KK}(T,K)=-e^{-rT}[-p(T,K)]
=e^{-rT}p(T,K).}
\]
Thus \(C_K\leq0\) and \(C_{KK}\geq0\): calls decrease and are convex in strike.
\end{frame}
% END INLINE: Dupire_Strike_Derivatives

\begin{frame}{Differentiate again: convexity becomes a density}

    \[\boxed{C_{KK}(T,K)=e^{-rT}p(T,K)\geq0.}\]
    \begin{block}{One relation, three interpretations}
    \begin{itemize}
    \item Static arbitrage: call prices must be convex in strike.
    \item Probability: curvature reveals the risk-neutral terminal density.
    \item Calibration: this curvature will enter the local variance.
    \end{itemize}\end{block}
    \begin{pausebox}Is this density an estimate of historical return frequencies?\end{pausebox}
    \begin{answerbox}No. It is implied by prices under \(\Q\).\end{answerbox}
\end{frame}

\begin{frame}{The density evolves according to a forward equation}

    Define \(a(T,s)=\sigma_{\rm loc}^2(T,s)s^2\). The forward equation is
    \[p_T=-\partial_s(rsp)+\tfrac12\partial_{ss}(ap).\]
    Differentiate the discounted call expectation:
    \[C_T=-rC+e^{-rT}\int_K^\infty(s-K)p_T(T,s)\dd s.\]
    \begin{block}{The calculation to perform}
    Insert the forward equation, then integrate by parts once for the drift
    term and twice for the diffusion term. Assume smoothness and vanishing tail terms.
    \end{block}
\end{frame}

% BEGIN INLINE: Dupire_Derivation
% Derivation in the no-dividend model, with constant interest rate r.
\begin{frame}{Derive the forward equation (1/2): start with It\^o}
\small
Put \(a(t,x)=\sigma_{\rm loc}^2(t,x)x^2\). For a smooth test function
\(\varphi\) with compact support in \((0,\infty)\), It\^o gives
\[
\dd\varphi(S_t)=\left[rS_t\varphi'(S_t)
+\tfrac12 a(t,S_t)\varphi''(S_t)\right]\dd t
+\sigma_{\rm loc}(t,S_t)S_t\varphi'(S_t)\dd W_t^\Q.
\]
Take expectations; the stochastic integral has mean zero:
\[
\frac{\dd}{\dd T}\E^\Q[\varphi(S_T)]
=\E^\Q\!\left[rS_T\varphi'(S_T)
+\tfrac12 a(T,S_T)\varphi''(S_T)\right].
\]
Write both sides using the density:
\[
\boxed{\int_0^\infty\varphi(x)p_T(T,x)\dd x
=\int_0^\infty\left[rx\varphi'(x)+\tfrac12a(T,x)\varphi''(x)\right]p(T,x)\dd x.}
\]
We assume the smoothness and integrability needed for these operations.
\end{frame}

\begin{frame}{Derive the forward equation (2/2): move the derivatives}
\small
Compact support removes the boundary terms at \(0\) and \(\infty\).
Integrate once for the drift and twice for the diffusion:
\[
\int_0^\infty rxp\,\varphi'\dd x
=-\int_0^\infty\varphi\,\partial_x(rxp)\dd x,
\]
\[
\int_0^\infty ap\,\varphi''\dd x
=-\int_0^\infty\partial_x(ap)\varphi'\dd x
=\int_0^\infty\varphi\,\partial_{xx}(ap)\dd x.
\]
The preceding equality holds for every such \(\varphi\), hence
\[
\boxed{p_T=-\partial_x(rxp)+\tfrac12\partial_{xx}(ap).}
\]
This evolves a probability density. Discounting enters separately when we
differentiate the option price.
\end{frame}

\begin{frame}{Differentiate in maturity: keep the discount derivative}
\small
With \(K\) fixed,
\[
\begin{aligned}
C_T(T,K)
&=-r e^{-rT}\int_K^\infty(x-K)p(T,x)\dd x
+e^{-rT}\int_K^\infty(x-K)p_T(T,x)\dd x\\
&=-rC+e^{-rT}(I_{\rm drift}+I_{\rm diff}),
\end{aligned}
\]
where the forward equation gives
\[
I_{\rm drift}=\int_K^\infty(x-K)[-\partial_x(rxp)]\dd x,
\]
\[
I_{\rm diff}=\tfrac12\int_K^\infty(x-K)\partial_{xx}(ap)\dd x.
\]
We now compute these two integrals, displaying the boundary terms.
Assume the tail terms shown on the next slides vanish.
\end{frame}

\begin{frame}{The drift integral: one integration by parts}
\small
Write \(m(x)=rxp(T,x)\). Then
\[
\begin{aligned}
I_{\rm drift}
&=-\int_K^\infty(x-K)m'(x)\dd x\\
&=-\big[(x-K)m(x)\big]_K^\infty
+\int_K^\infty m(x)\dd x.
\end{aligned}
\]
\begin{itemize}
\item At \(x=K\): \((K-K)m(K)=0\).
\item At infinity: assume \((x-K)m(x)\to0\).
\end{itemize}
\[
\boxed{I_{\rm drift}=r\int_K^\infty xp(T,x)\dd x.}
\]
The minus sign in the forward drift disappears after integration by parts.
\end{frame}

\begin{frame}{Express the tail first moment using call prices}
\small
Split \(x=(x-K)+K\):
\[
\begin{aligned}
e^{-rT}\int_K^\infty xp(T,x)\dd x
&=e^{-rT}\int_K^\infty(x-K)p(T,x)\dd x
+Ke^{-rT}\int_K^\infty p(T,x)\dd x\\
&=C(T,K)-K C_K(T,K).
\end{aligned}
\]
The sign follows from
\[
C_K(T,K)=-e^{-rT}\int_K^\infty p(T,x)\dd x.
\]
Therefore the discounted drift contribution is
\[
\boxed{e^{-rT}I_{\rm drift}=r(C-KC_K).}
\]
\end{frame}

\begin{frame}{The diffusion integral (1/2): remove one derivative}
\small
Write \(A(x)=a(T,x)p(T,x)\). The first integration by parts gives
\[
\begin{aligned}
I_{\rm diff}
&=\tfrac12\int_K^\infty(x-K)A''(x)\dd x\\
&=\tfrac12\big[(x-K)A'(x)\big]_K^\infty
-\tfrac12\int_K^\infty A'(x)\dd x.
\end{aligned}
\]
\begin{itemize}
\item At \(x=K\): the factor \(x-K\) again gives zero.
\item At infinity: assume \((x-K)A'(x)\to0\).
\end{itemize}
\[
\boxed{I_{\rm diff}=-\tfrac12\int_K^\infty A'(x)\dd x.}
\]
One boundary contribution remains to be evaluated.
\end{frame}

\begin{frame}{The diffusion integral (2/2): the boundary at the strike}
\small
The remaining integral is a boundary difference. Assume \(A(x)\to0\):
\[
I_{\rm diff}=-\tfrac12\big[A(x)\big]_K^\infty
=-\tfrac12(0-A(K))
=\tfrac12a(T,K)p(T,K).
\]
Substitute
\[
a(T,K)=\sigma_{\rm loc}^2(T,K)K^2,\qquad
e^{-rT}p(T,K)=C_{KK}(T,K).
\]
Then
\[
\boxed{e^{-rT}I_{\rm diff}
=\tfrac12\sigma_{\rm loc}^2(T,K)K^2 C_{KK}(T,K).}
\]
This is why the coefficient is evaluated at spot \(K\): the integration
leaves a boundary term at the call payoff's kink.
\end{frame}

\begin{frame}{Assemble Dupire: the two terms in \(rC\) cancel}
\small
Return to the maturity derivative:
\[
\begin{aligned}
C_T
&=-rC+e^{-rT}I_{\rm drift}+e^{-rT}I_{\rm diff}\\
&=-rC+r(C-KC_K)+\tfrac12\sigma_{\rm loc}^2(T,K)K^2 C_{KK}\\
&=-rK C_K+\tfrac12\sigma_{\rm loc}^2(T,K)K^2 C_{KK}.
\end{aligned}
\]
Thus
\[
\boxed{C_T+rK C_K=\tfrac12\sigma_{\rm loc}^2(T,K)K^2 C_{KK}.}
\]
When \(C_{KK}>0\), this equation identifies the local variance from
derivatives of today's call-price surface.
\source{R. Kohn, \href{https://math.nyu.edu/faculty/kohn/cont-time-finance/section9.pdf}{Continuous Time Finance, Section 9}: forward pricing equation and Dupire.}
\end{frame}
% END INLINE: Dupire_Derivation

\begin{frame}{Integration by parts gives the Dupire equation}

    The two integrals become
    \[\int_K^\infty(s-K)[-\partial_s(rsp)]\dd s
       =r\int_K^\infty s p(T,s)\dd s,\]
    \[\tfrac12\int_K^\infty(s-K)\partial_{ss}(ap)\dd s
       =\tfrac12 a(T,K)p(T,K).\]
    Since \(e^{-rT}\int_K^\infty s p\dd s=C-KC_K\),
    \[\boxed{C_T=-rK C_K+\tfrac12\sigma_{\rm loc}^2(T,K)K^2C_{KK}.}\]
    \begin{keybox}The forward equation acts on maturity and strike;
    the backward pricing PDE acts on valuation time and current spot.\end{keybox}
\end{frame}

\begin{frame}{Solve for the local variance}

    \[\boxed{\sigma_{\rm loc}^2(T,K)
        =\frac{C_T(T,K)+rKC_K(T,K)}{\tfrac12K^2C_{KK}(T,K)}.}\]
    \begin{itemize}
    \item Inputs are a sufficiently smooth, arbitrage-consistent call surface.
    \item The denominator must be positive where the formula is used.
    \item The result defines a model coefficient at time \(T\), spot \(K\).
    \end{itemize}
    \begin{watchbox}This expression assumes constant \(r\) and no dividends.
    It is not the formula for arbitrary carry or stochastic interest rates.\end{watchbox}
\end{frame}

% BEGIN INLINE: Dupire_Practice
\begin{frame}{Check Dupire on a Black--Scholes price surface}
\small
Use the usual \(d_1,d_2\), with \(\phi,\Phi\) the standard normal density and CDF.
For constant volatility \(\sigma\),
\[
\begin{aligned}
C_K&=-e^{-rT}\Phi(d_2),\\
C_{KK}&=\frac{e^{-rT}\phi(d_2)}{K\sigma\sqrt T},\\
C_T&=\frac{\sigma S_0\phi(d_1)}{2\sqrt T}+rKe^{-rT}\Phi(d_2).
\end{aligned}
\]
Using \(S_0\phi(d_1)=Ke^{-rT}\phi(d_2)\),
\[
C_T+rKC_K=\frac{\sigma S_0\phi(d_1)}{2\sqrt T}
=\tfrac12\sigma^2K^2C_{KK}.
\]
\[
\boxed{\sigma_{\rm loc}^2(T,K)=\sigma^2.}
\]
A flat implied-volatility surface recovers the constant BS coefficient.
\end{frame}


% END INLINE: Dupire_Practice

\begin{frame}{Differentiating noisy quotes is a numerical problem}

    For equally spaced strikes,
    \[C_{KK}(T,K)\approx
      \frac{C(T,K+h)-2C(T,K)+C(T,K-h)}{h^2}.\]
    \begin{pausebox}If each quote has independent error of standard deviation
    \(\varepsilon\), what is the standard deviation of this second difference?\end{pausebox}
    \begin{answerbox}
    \[\frac{\sqrt{1^2+(-2)^2+1^2}\,\varepsilon}{h^2}
      =\frac{\sqrt6\,\varepsilon}{h^2}.\]
    A smaller strike spacing amplifies quote noise. Fit a smooth,
    arbitrage-consistent surface before applying Dupire.
    \end{answerbox}
\end{frame}

\begin{frame}{Fitting today does not determine realistic future smile motion}

    \begin{block}{What a full vanilla surface determines}
    The risk-neutral marginal distribution of \(S_T\) for each maturity \(T\).
    \end{block}
    \begin{block}{What a path-dependent product also needs}
    The joint evolution of the stock between dates: averages, maxima, and
    barrier-crossing events depend on more than separate terminal marginals.
    \end{block}
    \begin{keybox}Local volatility supplies one dynamic model consistent with
    those marginals, under suitable assumptions. Other dynamics can imply
    different exotic prices and hedges.\end{keybox}
\end{frame}

\section{3. Stochastic volatility: price a barrier by replication}

\begin{frame}{Heston: the variance is a second random factor}
\small
Under the historical measure \(\Pp\), with \(Z_1,Z_2\) independent Brownian motions,
\[
\frac{\dd S_t}{S_t}=\mu\dd t+\sqrt{V_t}\dd W_t^S,\qquad
\dd V_t=\kappa_\Pp(\theta_\Pp-V_t)\dd t+\eta\sqrt{V_t}\dd W_t^V,
\]
\[
W^S=Z_1,\qquad W^V=\rho Z_1+\sqrt{1-\rho^2}\,Z_2,\qquad \dd\langle W^S,W^V\rangle_t=\rho\dd t.
\]
\begin{center}\begin{tabular}{clcl}
\toprule \(V_t\)&current variance (state)&\(\theta_\Pp\)&long-run variance\\
\(\kappa_\Pp\)&mean-reversion speed&\(\eta\)&volatility of variance\\
\(\rho\)&stock--variance correlation&\(\mu\)&stock drift\\\bottomrule
\end{tabular}\end{center}
\begin{keybox}The state is \((S_t,V_t)\): two dates with the same spot can carry different
option prices. Two Brownian shocks, but only one traded stock.\end{keybox}
\end{frame}


\begin{frame}{A variance move shifts the implied-volatility surface}
\small
In Heston, a vanilla price is \(C(t,S_t,V_t;K,T)\). Each implied volatility is
therefore a function \(\Sigma(t,S_t,V_t;K,T)\): when \(V_t\) moves, the whole surface moves.
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{tikzpicture}
\begin{axis}[width=\linewidth,height=4.0cm,axis lines=left,
xmin=0,xmax=3,ymin=10,ymax=32,xlabel={Maturity \(T-t\) (years)},
ylabel={ATM implied vol.\ (\%)},tick label style={font=\scriptsize},
label style={font=\scriptsize},grid=major,grid style={gray!15},
legend style={font=\scriptsize,draw=none,fill=white},legend pos=north east]
\addplot[CourseRed,thick,domain=0.02:3,samples=80]{100*sqrt(.04+(.09-.04)*(1-exp(-2*x))/(2*x))};
\addlegendentry{\(V_t=0.09\)}
\addplot[CourseBlue,very thick,domain=0.02:3,samples=80]{100*sqrt(.04+(.04-.04)*(1-exp(-2*x))/(2*x))};
\addlegendentry{\(V_t=0.04\)}
\addplot[CourseGreen,thick,dashed,domain=0.02:3,samples=80]{100*sqrt(.04+(.0225-.04)*(1-exp(-2*x))/(2*x))};
\addlegendentry{\(V_t=0.0225\)}
\end{axis}
\end{tikzpicture}
\column{0.47\textwidth}
\begin{itemize}
\item Under \(\Pp\), \(V\) has its own shock \(Z_2\). The surface can move
when \(S\) is unchanged.
\item The short end moves most. In the plotted guide, the long end tends to \(\sqrt\theta\).
\item In local volatility, the future surface is a deterministic function of
\((t,S_t)\). Heston adds a separate surface risk.
\end{itemize}
\end{columns}
\begin{keybox}The stock cannot hedge this move: we must price it, then hedge it with an option.\end{keybox}
\source{ATM implied variance \(\simeq\frac{1}{T-t}\int_t^T\E^\Q[V_u\mid V_t]\dd u\),
exact as \(\eta\to0\); \(\kappa=2\), \(\theta=0.04\).}
\end{frame}

\begin{frame}{It\^o along \(\Pp\)-paths: a mixed second derivative appears}
\small
For a smooth price \(f(t,S_t,V_t)\): \((\dd S)^2=VS^2\dd t\), \((\dd V)^2=\eta^2V\dd t\),
\(\dd S\dd V=\rho\eta SV\dd t\).
\begin{pausebox}What is the coefficient of \(f_{sv}\) in the drift of \(f\)?\end{pausebox}
\begin{answerbox}\(\rho\eta sv\): the two cross terms \(\tfrac12f_{sv}\) and
\(\tfrac12f_{vs}\) add up, so there is no factor \(1/2\).\end{answerbox}
\[
\dd f=\mathcal A^\Pp f\dd t+\sqrt{V_t}\,S_tf_s\dd W_t^S+\eta\sqrt{V_t}\,f_v\dd W_t^V,
\]
\[
\mathcal A^\Pp f=f_t+\mu sf_s+b_\Pp(v)f_v
+\tfrac12vs^2f_{ss}+\rho\eta svf_{sv}+\tfrac12\eta^2vf_{vv},
\qquad b_\Pp(v)=\kappa_\Pp(\theta_\Pp-v).
\]
No expectation, no pricing measure: an identity along the actual paths of \(S\) and \(V\).
\end{frame}

\begin{frame}{With the stock alone, the variance shock stays}
\small
Sell one claim \(u(t,S_t,V_t)\) and hold \(\Delta\) shares. The Brownian part of
\(\Delta\dd S-\dd u\) is
\[
\sqrt v\bigl[s(\Delta-u_s)-\rho\eta\,u_v\bigr]\dd Z_1
-\sqrt{1-\rho^2}\,\eta\sqrt v\,u_v\dd Z_2.
\]
\begin{pausebox}For \(|\rho|<1\) and \(\eta u_v\ne0\), which \(\Delta\) removes all the risk?\end{pausebox}
\begin{answerbox}None. \(\Delta=u_s+\rho\eta u_v/s\) cancels the \(Z_1\) term, but the
\(Z_2\) term does not involve \(\Delta\). Stock and cash cannot replicate the claim:
the market is incomplete.\end{answerbox}
\begin{keybox}Consequence for pricing: the risk-neutral measure is no longer unique.\end{keybox}
\end{frame}

\begin{frame}{From \(\Pp\) to \(\Q\): the stock fixes \(\lambda_1\)}
\small
Take an equivalent measure \(\Q\) with \(\dd Z_{j,t}^\Q=\dd Z_{j,t}^\Pp+\lambda_{j,t}\dd t\),
\(j=1,2\). The stock equation becomes
\[
\frac{\dd S_t}{S_t}=(\mu-\sqrt{V_t}\lambda_{1,t})\dd t
+\sqrt{V_t}\dd Z_{1,t}^\Q.
\]
The discounted stock is a \(\Q\)-martingale only if
\[
\boxed{\lambda_{1,t}=\frac{\mu-r}{\sqrt{V_t}}.}
\]
\begin{itemize}
\item \(\lambda_1\) is the market price of stock risk, exactly as in Black--Scholes.
\item Nothing in the stock constrains \(\lambda_2\), the price of the independent shock \(Z_2\).
\end{itemize}
These formulas are written on \(V_t>0\), with \(|\rho|<1\).
\end{frame}

\begin{frame}{The stock does not determine the variance drift under \(\Q\)}
\small
Substitute the same Brownian shifts into the variance equation:
\[
\dd V_t=\bigl[b_\Pp(V_t)-\pi\bigr]\dd t
+\eta\sqrt{V_t}\bigl(\rho\dd Z_{1,t}^\Q+\sqrt{1-\rho^2}\dd Z_{2,t}^\Q\bigr),
\]
\[
\boxed{\pi:=b_\Pp-b_\Q=\eta\sqrt{V_t}\bigl(\rho\lambda_{1,t}+\sqrt{1-\rho^2}\,\lambda_{2,t}\bigr)}
\quad\text{the variance risk premium.}
\]
\begin{block}{One traded stock, two independent shocks}
The stock fixes \(\lambda_1\) and leaves \(\lambda_2\) free: the market is incomplete, and
each choice of \(\lambda_2\) gives a different \(\Q\), hence a different \(b_\Q\).
\end{block}
\begin{itemize}
\item \(V\) is not traded: there is no ``drift \(=r\)'' rule for it. Its \(\Q\)-drift is the one
that makes discounted prices of traded assets depending on \(V\) (options) martingales.
\item It will be determined by option prices, through the model.
\item The diffusion term is unchanged: \(\eta\) and \(\rho\) are the same under \(\Pp\) and \(\Q\).
\end{itemize}
\end{frame}

\begin{frame}{An affine variance premium preserves the Heston form}
\small
\textbf{Model assumption:} the premium is affine, \(\pi(v)=\lambda v\). Then
\[
b_\Q(v)=\kappa_\Pp\theta_\Pp-(\kappa_\Pp+\lambda)v=\kappa_\Q(\theta_\Q-v),\qquad
\boxed{\kappa_\Q=\kappa_\Pp+\lambda,\quad
\theta_\Q=\frac{\kappa_\Pp\theta_\Pp}{\kappa_\Pp+\lambda},}
\]
with \(\kappa_\Pp+\lambda>0\).
\begin{itemize}
\item Why this assumption: \(V\) stays a Heston process under \(\Q\), so calls have a
closed form.
\item It imposes the constraint \(\kappa_\Q\theta_\Q=\kappa_\Pp\theta_\Pp\).
\item Sign convention \(b_\Q=b_\Pp-\lambda v\); \(\lambda\) need not be positive.
\end{itemize}
\begin{exampleblock}{Example used in the case study}
\(\kappa_\Pp=1.6,\ \theta_\Pp=0.05,\ \lambda=0.4\ \Longrightarrow\
\kappa_\Q=2,\ \theta_\Q=0.04\), and \(1.6\times0.05=2\times0.04=0.08\).
\end{exampleblock}
\source{Heston (1993), \href{https://finance.martinsewell.com/stylized-facts/volatility/Heston1993.pdf}{A Closed-Form Solution for Options with Stochastic Volatility}.}
\end{frame}

\begin{frame}{The Heston pricing PDE under the chosen \(\Q\)}
\small
The discounted claim value is a \(\Q\)-martingale. It\^o with the \(\Q\)-drifts gives
\[\boxed{\begin{aligned}
0={}&u_t+rsu_s+\kappa_\Q(\theta_\Q-v)u_v+\tfrac12vs^2u_{ss}\\
   &+\rho\eta svu_{sv}+\tfrac12\eta^2vu_{vv}-ru.
\end{aligned}}\]
For a European call, \(u(T,s,v)=(s-K)^+\).
\begin{watchbox}So far \(\Q\) is postulated: stock and cash alone do not select \(\lambda_2\).
The case study derives the same PDE by replication, once a liquid option is added
to the market.\end{watchbox}
\end{frame}

\begin{frame}{Case study: sell a barrier option, hedge it with a liquid call}
\small
\begin{block}{The claim we sell: a down-and-out call}
It pays \((S_T-K)^+\) at \(T\) if \(\min_{0\le t\le T}S_t>H\), and nothing otherwise
(continuous monitoring, no rebate). Here \(K=100\), \(H=85\), \(T=1\), \(S_0=100\), \(V_0=0.04\).
\end{block}
\begin{columns}[T,onlytextwidth]
\column{0.40\textwidth}
\textbf{Hedge instruments}
\begin{itemize}
\item the stock \(S\);
\item a liquid call \(g\): \(K_g=100\), \(T_g=1.5\);
\item cash (\(r=0\) here).
\end{itemize}
The call outlives the claim; assume \(g_v\ne0\) throughout the hedge.
\column{0.56\textwidth}
\textbf{What we know}\\[.4em]
\begin{tabular}{lcccc}
\toprule & \(\kappa\) & \(\theta\) & \(\eta\) & \(\rho\)\\\midrule
\(\Pp\): history of \(S,V\) & 1.6 & 0.05 & 0.4 & \(-0.6\)\\
\(\Q\): call surface & 2.0 & 0.04 & 0.4 & \(-0.6\)\\\bottomrule
\end{tabular}\\[.4em]
\(\eta,\rho\) agree: Girsanov changes drifts only.
\end{columns}
\begin{watchbox}Calls are liquid and trade at their Heston \(\Q\)-prices (calibrated to the whole
surface). Trading is continuous and costless. \(V_t\) is known.\end{watchbox}
\end{frame}

\begin{frame}{Step 1: write every price change along the \(\Pp\)-paths}
\small
By It\^o under \(\Pp\), a smooth price \(f(t,S_t,V_t)\) (the claim \(u\) before the barrier is hit,
or the call \(g\)) satisfies
\[
\dd f=\mathcal A^\Pp f\dd t+\sqrt{V_t}\,S_tf_s\dd W_t^S+\eta\sqrt{V_t}\,f_v\dd W_t^V,
\]
with \(W^S=Z_1^\Pp\), \(W^V=\rho Z_1^\Pp+\sqrt{1-\rho^2}\,Z_2^\Pp\), and \(\mathcal A^\Pp\) the
generator under \(\Pp\) (drifts \(\mu\) and \(b_\Pp\)).

Hold \(\Delta_t\) shares, \(\alpha_t\) calls, and cash \(b_t\). The portfolio is self-financing:
\[
X_t=\Delta_tS_t+\alpha_tg_t+b_t,\qquad
\dd X_t=\Delta_t\dd S_t+\alpha_t\dd g_t+rb_t\dd t.
\]
\begin{keybox}No expectation has been taken. These are identities along the paths that
the stock and the variance actually follow, under the historical measure.\end{keybox}
\end{frame}

\begin{frame}{Step 2: match both Brownian exposures}
\small
As in Black--Scholes, look for a self-financing \((\Delta,\alpha,b)\) with
\(X_t=u(t,S_t,V_t)\) for all \(t<\tau\wedge T\), where \(\tau=\inf\{t:\ S_t\le H\}\).
At \(\tau\) both are \(0\) and we liquidate; at \(T\), \(X_T=(S_T-K)^+\). With Step 1:
\[
\begin{aligned}
\dd X_t&=\bigl[\Delta\mu s+\alpha\mathcal A^\Pp g+rb\bigr]\dd t
+\sqrt v\,s\,(\Delta+\alpha g_s)\dd W^S+\eta\sqrt v\,\alpha g_v\dd W^V,\\
\dd u_t&=\mathcal A^\Pp u\dd t+\sqrt v\,s\,u_s\dd W^S+\eta\sqrt v\,u_v\dd W^V.
\end{aligned}
\]
\(X=u\) forces \(\dd X=\dd u\): match \(Z_2^\Pp\), then \(Z_1^\Pp\)
(\(|\rho|<1\), \(g_v\ne0\)), then the portfolio value:
\[
\boxed{\alpha=\frac{u_v}{g_v},\qquad \Delta=u_s-\alpha g_s,\qquad b=u-\Delta s-\alpha g.}
\]
\begin{pausebox}The holdings are fixed. What does matching the \(\dd t\) terms say?\end{pausebox}
\begin{answerbox}Nothing is left to choose: it is a condition on \(u\), the pricing equation.\end{answerbox}
\end{frame}

\begin{frame}{Step 3: the drift condition gives one variance premium \(\pi\)}
\small
Matching the \(\dd t\) terms:\quad
\(\Delta\mu s+\alpha\mathcal A^\Pp g+rb=\mathcal A^\Pp u.\)
Substitute \(\Delta=u_s-\alpha g_s\) and \(b=u-\Delta s-\alpha g\):
\[
\bigl[\mathcal A^\Pp u-ru-(\mu-r)su_s\bigr]
=\alpha\bigl[\mathcal A^\Pp g-rg-(\mu-r)sg_s\bigr].
\]
Since \(\alpha=u_v/g_v\), where \(u_v\ne0\) divide by \(u_v\):
\[
\boxed{\frac{\mathcal A^\Pp u-ru-(\mu-r)su_s}{u_v}
=\frac{\mathcal A^\Pp g-rg-(\mu-r)sg_s}{g_v}=:\pi(t,s,v).}
\]
\begin{itemize}
\item \(\pi\) is unique: the same identity holds for every claim replicated with \(S\) and \(g\).
\item It is the variance risk premium of the measure change:
\(\pi=\eta\sqrt v\,(\rho\lambda_1+\sqrt{1-\rho^2}\,\lambda_2)\).
\item Call prices, through the model, give \(b_\Q=b_\Pp-\pi\), and that is all the PDE uses.
\end{itemize}
\end{frame}

\begin{frame}{What call prices tell us: the \(\Q\)-drift, not \(\pi\)}
\small
The call trades at its Heston \(\Q\)-price, so \(g\) solves
\[
g_t+rsg_s+\kappa_\Q(\theta_\Q-v)g_v+\tfrac12vs^2g_{ss}+\rho\eta svg_{sv}
+\tfrac12\eta^2vg_{vv}-rg=0.
\]
Inject \(g\) into the right side of Step 3: the \(\mu\)-terms and all derivatives except \(g_v\) cancel,
\[
\pi(v)=\kappa_\Pp(\theta_\Pp-v)-\kappa_\Q(\theta_\Q-v)=0.4\,v\quad\text{here}
\quad(1.6\times0.05=2\times0.04=0.08).
\]
\begin{itemize}
\item Calibrating to the surface (several strikes and maturities) gives \(\kappa_\Q,\theta_\Q\).
One quoted price gives neither the function \(g(t,s,v)\) nor \(\pi\); the latter also needs \(\kappa_\Pp,\theta_\Pp\).
\item The model assumption turns today's cross-section of prices into a \(g\) valid in every state.
\item One call suffices to \emph{complete} the market for hedging; other calls are then redundant.
\end{itemize}
\begin{keybox}The PDE needs \(b_\Q\). Estimating \(\pi=b_\Pp-b_\Q\) separately
also needs historical estimates of \(\kappa_\Pp,\theta_\Pp\).\end{keybox}
\end{frame}

\begin{frame}{Step 4: replication gives the \(\Q\)-pricing PDE}
\small
Rearranging the identity for the claim gives
\[
u_t+rsu_s+\bigl[\kappa_\Pp(\theta_\Pp-v)-\pi\bigr]u_v
+\tfrac12vs^2u_{ss}+\rho\eta svu_{sv}+\tfrac12\eta^2vu_{vv}-ru=0.
\]
With the \(\Q\)-drift read from call prices, the bracket equals \(\kappa_\Q(\theta_\Q-v)\):
the Heston PDE under \(\Q\), now derived by replication.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Barrier claim \(u\)}
On \(s>H\), \(t<T\):\quad \(u(t,H,v)=0\) (absorbing),\\
\(u(T,s,v)=(s-K)^+\).
\end{block}
\column{0.48\textwidth}
\begin{block}{Hedge call \(g\)}
On \(s>0\), \(t<T_g\):\\
\(g(T_g,s,v)=(s-K_g)^+\).
\end{block}
\end{columns}
\begin{keybox}\(\mu\) has vanished, and so has \(\pi\): the PDE uses \(b_\Q\), and
\(\alpha=u_v/g_v\), \(\Delta=u_s-\alpha g_s\) use only derivatives of \(\Q\)-prices.
We hedge along \(\Pp\)-paths; we price with \(\Q\).\end{keybox}
\end{frame}

\section{4. Parametrizing the model: state, Q, and P}

% BEGIN INLINE: Model_Parametrization
\begin{frame}{Three kinds of inputs: a state, \(\Q\)-parameters, \(\Pp\)-parameters}
\small
\begin{center}
\begin{tabular}{@{}L{.12\textwidth}L{.19\textwidth}L{.20\textwidth}L{.39\textwidth}@{}}
\toprule Input & Role & Measure change & Where it comes from\\\midrule
\(S_t,V_t\) & state & yes: same paths & \(S_t\) observed; \(V_t\) refitted daily from options\\[.3em]
\(\eta,\rho\) & diffusion coefficients & yes & options; cross-check on time series\\[.3em]
\(\kappa,\theta\) & variance drift & no & \(\Q\): option prices; \(\Pp\): time series\\[.3em]
\(\mu\) & stock drift & \(\Pp\) only & not needed to price or to hedge\\[.3em]
\(r\) & discounting & --- & rates market\\\bottomrule
\end{tabular}
\end{center}
\begin{keybox}Prices and hedge ratios use \(\Q\)-parameters. Scenario probabilities,
risk limits, and the expected P\&L of an imperfect hedge use \(\Pp\).\end{keybox}
\end{frame}

\begin{frame}{Why \(\eta\) and \(\rho\) cannot change with the measure}
\small
Quadratic variations are computed path by path:
\[
\langle V\rangle_t=\int_0^t\eta^2V_u\dd u,\qquad
\langle\log S,V\rangle_t=\int_0^t\rho\eta V_u\dd u.
\]
They are limits of sums of squared (or cross) increments along the observed path.
Equivalent measures share the same almost-sure events, so these limits agree under
\(\Pp\) and \(\Q\). Girsanov changes drifts only.
\begin{pausebox}Options calibrate \(\rho=-0.6\); a time-series estimate gives \(\rho=-0.3\).
Is this a legitimate \(\Pp\)-versus-\(\Q\) difference?\end{pausebox}
\begin{answerbox}Not within Heston. Either the estimate is noisy (\(V\) is not observed,
and its proxies add error), or the model is misspecified. The barrier hedge relied on the true \(\rho\).\end{answerbox}
\end{frame}

\begin{frame}{Calibrating a model differs from estimating historical parameters}

    For vanilla quotes \(C_i^{\rm mkt}\), a price-based objective is
    \[\min_{\vartheta}\sum_iw_i
      \big(C^{\rm model}(K_i,T_i;\vartheta)-C_i^{\rm mkt}\big)^2,
      \quad\vartheta=(V_0,\kappa,\theta,\eta,\rho).\]
    \begin{itemize}
    \item State the instruments, weights, bid--ask ranges, and constraints.
    \item Check several parameter starting points and stability across days.
    \item Assess smile shape and maturities, not only one ATM option.
    \end{itemize}
    \begin{keybox}These are risk-neutral parameters fitted to prices.
    Historical dynamics under \(\Pp\) answer a different estimation question.\end{keybox}
\end{frame}

\begin{frame}{Level and term structure: \(V_0\), \(\theta\), \(\kappa\)}
\small
The expected average variance is exact under \(\Q\):
\[
w(T):=\frac1T\int_0^T\E^\Q[V_t]\dd t
=\theta+(V_0-\theta)\,\frac{1-e^{-\kappa T}}{\kappa T}.
\]
\(w(T)\) is the fair strike of a continuously sampled variance swap.
\begin{itemize}
\item \(T\to0\): \(w(T)\to V_0\); \quad \(T\to\infty\): \(w(T)\to\theta\).
\item \(\kappa\) sets the clock: the weight on \(V_0-\theta\) is \(1/2\) at \(\kappa T\approx1.6\).
\item \(V_0<\theta\): increasing \(w\); \quad \(V_0>\theta\): decreasing \(w\).
\end{itemize}
\begin{watchbox}\(\Sigma_{\rm ATM}^2(T)\approx w(T)\) is a deterministic-variance guide,
exact when \(\eta=0\). It is not an identity for stochastic variance.\end{watchbox}
\end{frame}

\begin{frame}{Term structure: work through the baseline}
\small
\begin{exampleblock}{Case study (\(\Q\)): \(V_0=\theta=0.04\), \(\kappa=2\)}
Expected average variance is flat at \(w(T)=0.04\), so the deterministic-variance
guide gives \(\sqrt{w(T)}=20\%\) at every maturity.
\end{exampleblock}
\begin{itemize}
\item With \(\eta=0.4\), the calculated ATM volatilities are about \(19.30\%\)
at three months and \(18.74\%\) at two years.
\item With \(V_0=0.09\), the one-year guide instead gives
\[
\sqrt{w(1)}=\sqrt{0.04+0.05\,\frac{1-e^{-2}}{2}}\simeq24.8\%.
\]
\end{itemize}
\begin{keybox}The mean of integrated variance helps interpret the level.
Its distribution and its dependence with stock shocks also affect option prices.\end{keybox}
\end{frame}

\begin{frame}{ATM skew: \(\rho\), \(\eta\), and maturity}
\small
Let \(k=\log(K/F_{0,T})\), with \(F_{0,T}=S_0e^{rT}\).
For \(V_0=\theta\), the leading small-\(\eta\) guide is
\[
\partial_k\Sigma^2(0,T)\approx\frac{\rho\eta}{\kappa T}
\left[1-\frac{1-e^{-\kappa T}}{\kappa T}\right].
\]
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Short maturities}
\[\partial_k\Sigma(0,T)\longrightarrow\frac{\rho\eta}{4\sqrt{V_0}}.\]
\end{block}
\column{0.48\textwidth}
\begin{block}{Large \(\kappa T\), within the guide}
\[\partial_k\Sigma^2(0,T)\approx\frac{\rho\eta}{\kappa T}.\]
\end{block}
\end{columns}
\begin{itemize}
\item \(\rho\) sets the sign; \(\eta\) scales the skew in this approximation.
\item Mean reversion damps the skew at longer maturities.
\end{itemize}
\source{R. Lee, \href{https://math.uchicago.edu/~rogerlee/impvol.pdf}{Implied Volatility: Statics, Dynamics, and Probabilistic Interpretation}, Example 3.9.}
\end{frame}

\begin{frame}{Skew: turn the guide into a numerical check}
\small
\begin{exampleblock}{Case study: \(\rho=-0.6\), \(\eta=0.4\), \(V_0=0.04\), \(\kappa=2\)}
At the short end,
\[
\partial_k\Sigma\approx\frac{-0.6\times0.4}{4\times0.2}=-0.30.
\]
Linearizing around \(20\%\), a 10\% OTM put (\(k=\log0.9\approx-0.105\))
has the guide \(20\%+(-0.30)(-0.105)\approx23.2\%\).
\end{exampleblock}
At \(T=1\), the variance-skew guide is \(-0.0681\); dividing by
\(2\times20\%\) gives a volatility slope of about \(-0.17\).

\begin{frame}{A calibration protocol you can defend}
\small
\begin{enumerate}
\item \textbf{Clean the quotes}: mids; no stale quotes, no calendar or butterfly arbitrage.
\item \textbf{Choose the error}: implied volatilities, or prices divided by BS vega, since
\(\bigl(C^{\rm model}-C^{\rm mkt}\bigr)/\mathcal V^{\rm BS}\simeq\Sigma^{\rm model}-\Sigma^{\rm mkt}\).
\item \textbf{Impose constraints}: \(V_0,\kappa,\theta,\eta>0\) and \(|\rho|<1\);
report the Feller ratio \(2\kappa\theta/\eta^2\).
\item \textbf{Start from the parameter guides}, then try several starting points.
\item \textbf{Validate}: residuals against bid--ask widths, out-of-sample strikes,
and stability from one day to the next.
\end{enumerate}
\begin{pausebox}Our baseline is \((\kappa,\theta,\eta)=(2,0.04,0.4)\). Compute \(2\kappa\theta/\eta^2\).\end{pausebox}
\begin{answerbox}\(2\times2\times0.04/0.16=1\): exactly on the Feller boundary.
Calibrated equity parameters often violate it. This is allowed, but the simulation
scheme must then handle \(V\) near zero (Section 5).\end{answerbox}
\end{frame}

\begin{frame}{Daily recalibration exposes what the model leaves out}
\small
Heston treats \(\kappa,\theta,\eta,\rho\) as constants, yet desks refit them every day.
Take the barrier hedge at \(t=0\) (\(-1\) claim, \(+0.8386\) calls, stock and cash),
then refit one parameter with \(S\) and \(V\) unchanged:
\begin{center}
\begin{tabular}{lrrr}
\toprule Refit & \(\Delta u\) & \(\alpha\,\Delta g\) & Hedger's P\&L \(=\alpha\Delta g-\Delta u\)\\\midrule
\(\eta\colon0.40\to0.45\) & \(-0.119\) & \(-0.100\) & \(+0.018\)\\
\(\rho\colon-0.6\to-0.7\) & \(-0.025\) & \(-0.048\) & \(-0.023\)\\\bottomrule
\end{tabular}
\end{center}
\begin{itemize}
\item The claim and the call load differently on \(\eta\) and \(\rho\); the model's
own hedge does not cover parameter moves.
\item Responses: hedge skew and wings with OTM options, hold reserves, or change model.
\end{itemize}
\begin{keybox}Calibration fixes today's prices. The hedge depends on the dynamics the
model assumes between today and maturity.\end{keybox}
\end{frame}

\section{5. From model equations to simulations}

\begin{frame}{Generate two correlated shocks from independent normals}

    At each time step draw independent \(Z_1,Z_2\sim N(0,1)\), then set
    \[\Delta W^S=\sqrt{\Delta t}\,Z_1,\quad
      \Delta W^V=\sqrt{\Delta t}\left(\rho Z_1+\sqrt{1-\rho^2}Z_2\right).\]
    \begin{pausebox}Check the variances and covariance.\end{pausebox}
    \begin{answerbox}
    \[\Var(\Delta W^S)=\Var(\Delta W^V)=\Delta t,\quad
       \Cov(\Delta W^S,\Delta W^V)=\rho\Delta t.\]
    Draw a fresh independent pair at each step and for each independent path.
    \end{answerbox}
\end{frame}

\begin{frame}{A positive continuous variance can have a negative Euler step}

    A naive Euler update is
    \[V_{n+1}=V_n+\kappa(\theta-V_n)\Delta t
                 +\eta\sqrt{V_n}\,\Delta W_n^V.\]
    \begin{block}{A problem for the next square root}
    A sufficiently negative normal shock makes \(V_{n+1}<0\).
    Continuous-time nonnegativity does not guarantee nonnegative Euler values.
    \end{block}
    \begin{watchbox}The Feller condition \(2\kappa\theta\geq\eta^2\)
    concerns the continuous CIR boundary. It does not repair the Euler scheme.\end{watchbox}
\end{frame}

\begin{frame}{Name the approximation: full truncation and log stock steps}

    Put \(V_n^+=\max(V_n,0)\), retain the raw \(V_n\), and use
    \[\begin{aligned}
    V_{n+1}&=V_n+\kappa(\theta-V_n^+)\Delta t
                     +\eta\sqrt{V_n^+}\,\Delta W_n^V,\\
    S_{n+1}&=S_n\exp\!\left[(r-\tfrac12V_n^+)\Delta t
                              +\sqrt{V_n^+}\Delta W_n^S\right].
    \end{aligned}\]
    \begin{keybox}The stock stays positive. Both updates still approximate
    Heston dynamics; convergence must be checked by refining \(\Delta t\).\end{keybox}
    \source{Full truncation: Lord, Koekkoek and van Dijk, \href{https://repub.eur.nl/pub/7738}{A Comparison of Biased Simulation Schemes for Stochastic Volatility Models}.}
\end{frame}

\begin{frame}{Exit questions: price, state, and hedge}

    \begin{enumerate}
    \item Does local volatility add an independent Brownian shock?
    \item Does fitting one ATM price identify all Heston parameters?
    \item Does fitting today's full vanilla surface determine all exotic prices?
    \end{enumerate}
    \begin{answerbox}
    \begin{enumerate}
    \item No: the coefficient is a function of time and spot.
    \item No: one scalar quote cannot generally identify five parameters.
    \item No: path-dependent claims also depend on joint dynamics.
    \end{enumerate}\end{answerbox}
\end{frame}
