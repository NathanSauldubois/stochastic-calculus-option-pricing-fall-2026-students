\begin{keyidea}{Goals and logical structure}
Lecture 8 gave model-free principles for comparing claims. Lecture 9 asks how
those principles discipline the prices of the options traders quote every day:
\[
\begin{aligned}
\text{payoffs of calls, puts, and combinations}
&\longrightarrow \text{put--call parity and model-free bounds}\\
&\longrightarrow \text{Black--Scholes prices, Greeks, and implied volatility}\\
&\longrightarrow \text{the P\&L of a hedge run with the wrong volatility}\\
&\longrightarrow \text{strike and calendar restrictions on the option surface.}
\end{aligned}
\]
Whenever a check fails, the goal is to write the explicit trade, not only the
violated inequality.
\end{keyidea}

\begin{assumptionbox}{Conventions used throughout}
European options on one share; no dividends unless stated otherwise.
\(\tau=T-t>0\) is in years, \(r\) is constant, \(D=e^{-r\tau}\), and
volatility is annualized.
\end{assumptionbox}

% ============================================================
\section{Payoffs and Market Language}
% ============================================================

\subsection{Calls, puts, moneyness, and time value}

\begin{definition}[Vanilla options]
A European \emph{call} (resp.\ \emph{put}) with strike \(K\) and maturity
\(T\) gives the right to buy (resp.\ sell) one share for \(K\) at \(T\).
Exercise occurs only at \(T\).
\[
\begin{array}{c|cc}
 &\text{Call}&\text{Put}\\ \hline
\text{Payoff at }T &(S_T-K)^+&(K-S_T)^+\\
\text{Intrinsic value at }t &(S_t-K)^+&(K-S_t)^+\\
\text{Time value at }t &C_t-(S_t-K)^+&P_t-(K-S_t)^+
\end{array}
\]
\end{definition}

A call is \emph{in the money} (ITM) when \(S_t>K\); a put is ITM when
\(S_t<K\); an option is \emph{at the money} (ATM) when \(S_t\approx K\).
Out-of-the-money (OTM) options have zero intrinsic value but may have a
positive premium. A short position reverses every payoff.

\begin{warningbox}{Time value need not be nonnegative}
At \(r=q=0\), time value is nonnegative and vanishes at expiry. With carry,
a European premium can be below spot intrinsic value.
\end{warningbox}

\subsection{Payoff combinations}

\begin{proposition}[Standard combinations]
For \(h>0\),
\[
\begin{aligned}
\text{Straddle:}\quad &(S_T-K)^++(K-S_T)^+=|S_T-K|,\\
\text{Protective put:}\quad &S_T+(K-S_T)^+=\max(S_T,K),\\
\text{Covered call:}\quad &S_T-(S_T-K)^+=\min(S_T,K),\\
\text{Call minus put:}\quad &(S_T-K)^+-(K-S_T)^+=S_T-K,\\
\text{Butterfly:}\quad &(S_T-K+h)^+-2(S_T-K)^++(S_T-K-h)^+
=(h-|S_T-K|)^+\ge0.
\end{aligned}
\]
\end{proposition}

The protective put keeps the upside but cannot finish below \(K\); the
covered call gives up the upside above \(K\) in exchange for the call premium.
A straddle buys a move without choosing its direction.

\begin{warningbox}{Payoff is not profit}
Profit also includes premiums and their financing. For a straddle at
\(r=0\), the break-even prices are \(K\pm(C_t+P_t)\).
\end{warningbox}

% ============================================================
\section{Put--Call Parity and Model-Free Bounds}
% ============================================================

\subsection{Dominance and parity}


\begin{theorem}[Put--call parity]
For matched European calls and puts (same underlying, \(K\), \(T\), and
settlement),
\[
\boxed{C_t+KD=P_t+S_t,\qquad C_t-P_t=S_t-KD.}
\]
With a known continuous dividend yield \(q\),
\[
C_t-P_t=S_te^{-q\tau}-Ke^{-r\tau}.
\]
\end{theorem}
\begin{proof}
The portfolio ``call \(+\) \(KD\) in cash'' and the portfolio
``put \(+\) one share'' both pay \(\max(S_T,K)\) at \(T\). By the law of one
price they have the same value at \(t\). With yield \(q\), hold
\(e^{-q\tau}\) shares, reinvesting dividends, to end with one share.
\end{proof}

\subsection{No-arbitrage bounds}

\begin{proposition}[Model-free bounds]
\[
\boxed{(S_t-KD)^+\le C_t\le S_t,\qquad (KD-S_t)^+\le P_t\le KD.}
\]
With continuous yield \(q\), replace \(S_t\) by \(S_te^{-q\tau}\) in these
bounds.
\end{proposition}
\begin{proof}
Upper bound: one share superhedges the call, \(S_T\ge(S_T-K)^+\). If
\(C_t>S_t\), selling the call and buying the share earns \(C_t-S_t>0\) today
with a nonnegative terminal payoff. Lower bound: parity gives
\(C_t=P_t+S_t-KD\ge S_t-KD\) since \(P_t\ge0\), and \(C_t\ge0\). The put
bounds follow in the same way.
\end{proof}

More generally, if liquid strategies satisfy \(Y_T\le H\le X_T\), then
\(Y_0\le\text{price}(H)\le X_0\). If \(H\) is too expensive, sell it and buy
the superhedge \(X\); if it is too cheap, buy it and short the subhedge \(Y\).

\begin{keyidea}{Bounds and implied volatility}
A quote outside its bounds has no nonnegative BS implied volatility.
A strictly interior quote has a unique finite, positive implied volatility.
\end{keyidea}

% ============================================================
\section{Black--Scholes as a Benchmark}
% ============================================================

\subsection{Model, pricing PDE, and vanilla formulas}

No-arbitrage alone gives relations and bounds, not generally a unique price.
Black--Scholes adds a model, hence a distribution for \(S_T\).

\begin{assumptionbox}{Black--Scholes model}
Under \(\Q\), with no dividends and constant \(r\) and \(\sigma>0\),
\[
\frac{\dd S_u}{S_u}=r\dd u+\sigma\dd W_u^{\Q},\qquad
\partial_t v+rs\,\partial_s v+\frac12\sigma^2s^2\partial_{ss}v-rv=0,
\quad v(T,s)=h(s).
\]
\end{assumptionbox}

\begin{theorem}[Black--Scholes formulas]
Let
\[
d_1=\frac{\log(S_t/K)+(r+\tfrac12\sigma^2)\tau}{\sigma\sqrt\tau},
\qquad d_2=d_1-\sigma\sqrt\tau,
\qquad \phi(z)=\Phi'(z)=\frac{e^{-z^2/2}}{\sqrt{2\pi}},
\]
where \(\Phi\) is the standard normal CDF. Then
\[
\boxed{C_t=S_t\Phi(d_1)-KD\Phi(d_2),\qquad
P_t=KD\Phi(-d_2)-S_t\Phi(-d_1).}
\]
Moreover \(\Phi(d_2)=\Q(S_T>K\mid\cF_t)\).
\end{theorem}

\subsection{ATM short-maturity limits}

\begin{proposition}[Square-root rule]
At \(S_t=K\), define \(a=S_t\sigma\sqrt\tau/\sqrt{2\pi}\). For fixed
\(r,\sigma\), as \(\tau\downarrow0\),
\[
C_t=a+\frac{rS_t}{2}\tau+O(\tau^{3/2}),\qquad
P_t=a-\frac{rS_t}{2}\tau+O(\tau^{3/2}).
\]
In particular,
\[
\boxed{C_t\sim P_t\sim\frac{S_t\sigma\sqrt\tau}{\sqrt{2\pi}}
\approx0.4S_t\sigma\sqrt\tau,\qquad
C_t+P_t\sim\sqrt{\frac2\pi}S_t\sigma\sqrt\tau.}
\]
\end{proposition}
\begin{proof}[Sketch]
At \(r=0\) and \(S_t=K\), \(C_t=P_t=S_t[2\Phi(\sigma\sqrt\tau/2)-1]\). Use
\(\Phi(x)=\tfrac12+x/\sqrt{2\pi}+O(x^3)\) as \(x\to0\).
\end{proof}

\begin{warningbox}{An ATM approximation only}
The square-root rule holds at the money. Away from the money, use the full
Black--Scholes formula.
\end{warningbox}

\subsection{Greeks and the Delta hedge}

\begin{definition}[Greeks]
For a price function \(v(t,s)\),
\[
\Delta=\partial_s v,\quad \Gamma=\partial_{ss}v,\quad
\nu=\partial_\sigma v,\quad \Theta=\partial_t v,\quad \rho=\partial_r v.
\]
\end{definition}

\begin{proposition}[Black--Scholes Greeks]
\[
\Delta_C=\Phi(d_1),\qquad \Delta_P=\Phi(d_1)-1,\qquad
\boxed{\Gamma_C=\Gamma_P=\frac{\phi(d_1)}{S_t\sigma\sqrt\tau}>0,}
\]
\[
\boxed{\nu_C=\nu_P=S_t\sqrt\tau\,\phi(d_1)>0,}\qquad
\rho_C=K\tau D\Phi(d_2),\quad \rho_P=-K\tau D\Phi(-d_2),
\]
\[
\begin{aligned}
\Theta_C&=-\frac{S_t\sigma\phi(d_1)}{2\sqrt\tau}-rKD\Phi(d_2),\\
\Theta_P&=-\frac{S_t\sigma\phi(d_1)}{2\sqrt\tau}+rKD\Phi(-d_2).
\end{aligned}
\]
\end{proposition}

Theta uses calendar time \(t\), not time to maturity \(\tau\). At \(r=0\),
\(\Theta=-\tfrac12\sigma^2S_t^2\Gamma\).

\begin{keyidea}{Delta hedging and Gamma}
Hedge a short option with \(+\Delta\) shares; hedge a long option with
\(-\Delta\) shares. For a small spot move,
\[
\delta v\approx\Delta\,\delta S,\qquad
\delta\Delta\approx\Gamma\,\delta S.
\]
Delta is the first-order hedge ratio; positive Gamma is why that hedge must be
rebalanced. After a rise in spot, a call's Delta increases: the call seller
buys more shares, while a long-call hedger sells. Delta-neutral is not
risk-free: Gamma, volatility, and discrete rehedging still matter.
\end{keyidea}

\begin{proposition}[ATM Greeks near expiry]
At \(S_t=K\),
\[
\Gamma_{\rm ATM}\sim\frac1{S_t\sigma\sqrt{2\pi\tau}},\qquad
\Delta_C\longrightarrow
\begin{cases}0&S_t<K,\\ \tfrac12&S_t=K,\\1&S_t>K,\end{cases}
\]
where the Delta limits are taken at fixed spot as \(\tau\downarrow0\).
At \(r=0\), \(\Delta_{\rm ATM}=\Phi(\sigma\sqrt\tau/2)>\tfrac12\) before expiry.
\end{proposition}

\begin{warningbox}{Pre-expiry limits versus the payoff kink}
These are pre-expiry limits at fixed spot. At \(S_T=K\), the terminal payoff
has a kink; its classical Delta and Gamma are undefined. Gamma concentrates
near the strike as expiry approaches and tends to zero at fixed \(S_t\ne K\).
\end{warningbox}

\begin{studentexercise}{Gamma and Vega of combinations}
(a) What are the Gamma and Vega of a matched call-minus-put position?
(b) Buy a \(K_1\)-strike call and sell a \(K_2\)-strike call, \(K_1<K_2\),
same expiry. Must the spread's Vega be positive?
\end{studentexercise}
\begin{shortsolution}
(a) Parity gives \(C_t-P_t=S_t-KD\), linear in spot and independent of
volatility: \(\Delta=1\), \(\Gamma=\nu=0\).
(b) \(\nu_{\rm spread}=\nu(K_1)-\nu(K_2)\) can have either sign. When spot is
well above both strikes, \(d_1(K_1)>d_1(K_2)>0\), so \(\nu(K_1)<\nu(K_2)\)
and the spread has negative Vega. A nonnegative payoff does not imply
positive Vega.
\end{shortsolution}

% ============================================================
\section{Volatility: Historical, Realized, and Implied}
% ============================================================

\subsection{Historical and realized annualized volatility}

\begin{definition}[Historical volatility]
From \(n+1\) observed prices (\(n\ge2\)), define log returns and their sample
mean and variance:
\[
R_i=\log\!\left(\frac{S_{t_i}}{S_{t_{i-1}}}\right),\qquad
\overline R=\frac1n\sum_{i=1}^nR_i,\qquad
s_R^2=\frac1{n-1}\sum_{i=1}^n(R_i-\overline R)^2.
\]
For \(A\) observations per year (\(A=252\) for daily trading returns),
\[
\boxed{\widehat\sigma_{\rm hist,ann}=\sqrt A\,s_R.}
\]
\end{definition}

The square-root annualization assumes uncorrelated returns with a common mean
and variance; normality is not required.

\begin{definition}[Realized volatility]
Over a window \(\Delta T=T-t=n/A\),
\[
\boxed{\mathrm{RV}_{\rm ann}=\frac1{\Delta T}\sum_{i=1}^nR_i^2,\qquad
\widehat\sigma_{\rm real,ann}=\sqrt{\mathrm{RV}_{\rm ann}}.}
\]
\end{definition}

Realized variance uses uncentered squared returns. For a continuous diffusion,
as sampling becomes finer,
\[
\sum_iR_i^2\xrightarrow{\Pp}\int_t^T\sigma_u^2\dd u,
\qquad \sigma_{\rm real,ann}^2=\frac1{T-t}\int_t^T\sigma_u^2\dd u.
\]

\subsection{Implied volatility and the surface}

\begin{definition}[Implied volatility]
The implied volatility \(\sigma_{{\rm imp},t}(K,T)\) solves
\[
\boxed{C^{\rm BS}(S_t,K,\tau,r,\sigma_{{\rm imp},t}(K,T))=C_t^{\rm mkt}(K,T).}
\]
At a fixed observation date \(t\), the map
\((K,T)\mapsto\sigma_{{\rm imp},t}(K,T)\) is the implied volatility surface.
\end{definition}

Strike slices form a smile or skew; expiry slices form a term structure.
Constant-volatility BS gives a flat implied surface.

\begin{warningbox}{A price, not a forecast}
Implied volatility is a price expressed in BS volatility units. It is neither
historical realized volatility nor a guaranteed forecast of future realized
volatility.
\end{warningbox}

\begin{studentexercise}{a put instead of a call}
You quoted a European call in implied volatility. The client wants the put
with the same \(K,T\). Should the volatility change? And what do traders mean
when they say that ``puts trade at higher vol''?
\end{studentexercise}
\begin{shortsolution}
No: matched, parity-consistent calls and puts have the same implied
volatility, because the BS formulas satisfy parity at every common
volatility, in or out of the money. The trader's statement compares
low-strike OTM puts with high-strike OTM calls, i.e.\ \emph{different
strikes}: a downside skew, which is consistent with parity.
\end{shortsolution}

% ============================================================
\section{Replication with the Wrong Volatility}
% ============================================================

\begin{assumptionbox}{Setting: seller's convention}
No dividends, constant \(r\), continuous and costless rebalancing. Actual
dynamics and a fixed-volatility model price:
\[
\frac{\dd S_u}{S_u}=\mu_u\dd u+\sigma_u\dd W_u^{\Pp},\qquad
v(u,s)=C^{\rm BS}(s,K,T-u,r,\widehat\sigma),\quad \widehat\sigma>0.
\]
Use model Greeks \(\Delta_u=v_s(u,S_u)\), \(\Gamma_u=v_{ss}(u,S_u)\),
evaluated with \(\widehat\sigma\). Sell the call and run
\[
V_0=v(0,S_0),\qquad
\dd V_u=\Delta_u\dd S_u+r(V_u-\Delta_uS_u)\dd u,\qquad
\Pi_u=V_u-v(u,S_u).
\]
\end{assumptionbox}

\begin{theorem}[Gamma-weighted P\&L]
\[
\boxed{\dd\Pi_u=r\Pi_u\dd u+
\frac12(\widehat\sigma^2-\sigma_u^2)S_u^2\Gamma_u\dd u,\quad \Pi_0=0,}
\]
and therefore
\[
\boxed{\Pi_T=V_T-(S_T-K)^+
=\frac12\int_0^T e^{r(T-u)}S_u^2\Gamma_u
          (\widehat\sigma^2-\sigma_u^2)\dd u.}
\]
\end{theorem}
\begin{proof}
By It\^o's formula under the actual dynamics,
\(\dd v=v_t\dd u+\Delta_u\dd S_u+\tfrac12\sigma_u^2S_u^2\Gamma_u\dd u\).
The model price solves the BS PDE with \(\widehat\sigma\):
\(v_t=rv-rS_u\Delta_u-\tfrac12\widehat\sigma^2S_u^2\Gamma_u\). Subtracting
from \(\dd V_u\), the \(\dd S_u\) terms cancel and
\[
\dd\Pi_u=r(V_u-v)\dd u+\tfrac12(\widehat\sigma^2-\sigma_u^2)S_u^2\Gamma_u\dd u.
\]
Then \(\dd(e^{-ru}\Pi_u)=\tfrac12e^{-ru}(\widehat\sigma^2-\sigma_u^2)
S_u^2\Gamma_u\dd u\); integrate from \(0\) to \(T\) and use
\(v(T,S_T)=(S_T-K)^+\).
\end{proof}

\begin{keyidea}{Reading the formula}
An initial premium mismatch adds \(e^{rT}\Pi_0\). The long-option hedge
reverses the sign. For calls and puts, \(\Gamma_u>0\): pointwise
overestimating volatility benefits the seller.
\end{keyidea}

\begin{warningbox}{Realized volatility alone does not decide the P\&L}
An unweighted realized variance does not determine this Gamma-weighted P\&L.
Discrete rehedging and costs add further effects.
\end{warningbox}

\begin{studentexercise}{realized above implied}
You buy a call at implied volatility \(\widehat\sigma\) and Delta-hedge it
daily. Realized volatility over the life of the option exceeds
\(\widehat\sigma\). Must you have made money?
\end{studentexercise}
\begin{shortsolution}
No. At \(r=0\), in the continuous costless benchmark,
\(\Pi_T^{\rm long}=\tfrac12\int_0^TS_t^2\Gamma_t(\sigma_t^2-\widehat\sigma^2)\dd t\).
The weights \(S_t^2\Gamma_t\) are absent from realized volatility; high
volatility far from the strike earns little. Daily rehedging and costs add
further differences.
\end{shortsolution}

% ============================================================
\section{Strike Restrictions at a Fixed Expiry}
% ============================================================

\begin{proposition}[Monotonicity, slope, and convexity in strike]
For \(K_1<K_2\), with \(D=e^{-r\tau}\),
\[
\boxed{0\le C_t(K_1)-C_t(K_2)\le D(K_2-K_1),\qquad
0\le P_t(K_2)-P_t(K_1)\le D(K_2-K_1).}
\]
For equally spaced strikes,
\[
\boxed{C_t(K-h)-2C_t(K)+C_t(K+h)\ge0.}
\]
If prices are smooth, \(-D\le\partial_K C\le0\), \(0\le\partial_KP\le D\),
and \(\partial_{KK}C=\partial_{KK}P\ge0\).
\end{proposition}
\begin{proof}
The call spread pays \((S_T-K_1)^+-(S_T-K_2)^+\in[0,K_2-K_1]\); price the two
bounds by dominance. The butterfly payoff \((h-|S_T-K|)^+\) is nonnegative,
so its price is nonnegative. Puts follow by parity.
\end{proof}

\begin{warningbox}{What this implies for the smile}
For implied quotes \(I_i=\sigma_{{\rm imp},t}(K_i,T)\), apply these
restrictions to
\[
C_i=C^{\rm BS}(S_t,K_i,\tau,r,I_i).
\]
They constrain the \(I_i\) jointly; implied volatility itself need not
decrease or be convex in strike.
\end{warningbox}

% ============================================================
\section{Calendar Restrictions}
% ============================================================

\subsection{Calendar spread at zero rates and dividends}

For the same strike and \(t<T_1<T_2\), buy the \(T_2\) call and sell the
\(T_1\) call. Close at \(T_1\) by selling the remaining long call and
settling the short:
\[
B_t=C_t(K,T_2)-C_t(K,T_1),\qquad
B_{T_1}=C_{T_1}(K,T_2)-(S_{T_1}-K)^+,\qquad
\boxed{\Pi_{T_1}=B_{T_1}-B_t.}
\]

\begin{proposition}[Calendar monotonicity]
At \(r=q=0\), \(B_{T_1}\ge0\), and hence
\[
\boxed{B_t=C_t(K,T_2)-C_t(K,T_1)\ge0.}
\]
\end{proposition}
\begin{proof}
The intrinsic-value lower bound at \(T_1\) gives
\(C_{T_1}(K,T_2)\ge(S_{T_1}-K)^+\); equivalently, since \(S\) is a
\(\Q\)-martingale, conditional Jensen yields
\(\E^\Q[(S_{T_2}-K)^+\mid\cF_{T_1}]\ge(S_{T_1}-K)^+\). If \(B_t<0\), selling
the short-dated call and buying the long-dated one earns \(-B_t>0\) today
with a nonnegative closing cashflow \(B_{T_1}\).
\end{proof}

\begin{warningbox}{No arbitrage is not a guaranteed profit}
A negative initial spread price is an arbitrage; a nonnegative initial price
does not guarantee a profit.
\end{warningbox}

\subsection{Calendar condition in implied-volatility units}

\begin{proposition}[Total implied variance increases]
At \(r=q=0\), fixed spot and strike, let \(\tau_i=T_i-t\),
\(I_i=\sigma_{{\rm imp},t}(K,T_i)\), and \(w_i=\tau_iI_i^2\). Then
\[
\boxed{C_t(K,T_2)\ge C_t(K,T_1)
\iff \tau_2I_2^2\ge\tau_1I_1^2
\iff I_2\ge I_1\sqrt{\frac{\tau_1}{\tau_2}}.}
\]
\end{proposition}
\begin{proof}
At \(r=q=0\), the BS call depends on \(\sigma\) and \(\tau\) only through
\(\sigma\sqrt\tau=\sqrt w\), and it is strictly increasing in it since
\(\nu>0\).
\end{proof}

\subsection{Deterministic carry: fix forward moneyness}

For known constant \(r,q\), define
\[
D(\tau)=e^{-r\tau},\quad F(\tau)=S_te^{(r-q)\tau},\quad
x=\frac K{F(\tau)},\quad
c(x,\tau)=\frac{C_t(xF(\tau),t+\tau)}{D(\tau)F(\tau)}.
\]

\begin{proposition}[Calendar condition with carry]
With \(w(x,\tau)=\tau\sigma_{{\rm imp},t}(xF(\tau),t+\tau)^2\),
\[
\boxed{c(x,\tau_2)\ge c(x,\tau_1)\iff w(x,\tau_2)\ge w(x,\tau_1).}
\]
\end{proposition}

\begin{warningbox}{Different strikes across expiries}
The comparison holds at fixed \(x\), generally with different strikes
\(K_i=xF(\tau_i)\) across expiries. Each maturity slice must also satisfy the
strike constraints of Section~6.
\end{warningbox}

\section{Synthesis: Diagnose Before You Price}
% ============================================================

\begin{keyidea}{A disciplined order for checking an option quote}
\begin{enumerate}
  \item Identify the exact contract: underlying, strike, maturity, dividends.
  \item Check positivity and the basic upper and lower bounds.
  \item Pair calls and puts with the same \(K,T\) to check parity.
  \item Across strikes, check monotonicity and convexity.
  \item Across maturities, apply calendar checks only with the right
        assumptions.
  \item If a check fails, write the portfolio before calling it an arbitrage.
\end{enumerate}
\end{keyidea}

\begin{keyidea}{Final synthesis}
\[
\boxed{\text{option payoff geometry}
\quad\Longrightarrow\quad
\text{arbitrage-free prices and hedges}.}
\]
Replication gives parity; dominance gives bounds; strike convexity gives
butterfly tests. Black--Scholes gives prices, Greeks, and implied
volatilities, but never replaces model-free consistency checks. A wrong
volatility creates a variance error weighted by \(S_u^2\Gamma_u\).
\end{keyidea}
